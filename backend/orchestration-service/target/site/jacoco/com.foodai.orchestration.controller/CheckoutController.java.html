package com.foodai.orchestration.controller;

import com.foodai.orchestration.domain.model.SagaType;
import com.foodai.orchestration.dto.request.CheckoutRequest;
import com.foodai.orchestration.dto.request.StartSagaRequest;
import com.foodai.orchestration.dto.response.ApiResponse;
import com.foodai.orchestration.dto.response.CheckoutResponse;
import com.foodai.orchestration.dto.response.SagaResponse;
import com.foodai.orchestration.service.SagaService;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.responses.ApiResponses;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.time.Instant;
import java.util.HashMap;
import java.util.Map;

/**
 * REST controller for checkout operations.
 *
 * @author FoodAI Team
 */
@RestController
@RequestMapping("/api/v1/checkout")
@RequiredArgsConstructor
@Slf4j
@Tag(name = "Checkout", description = "APIs for checkout operations")
public class CheckoutController {

    private final SagaService sagaService;

    /**
     * Initiates checkout flow.
     *
     * @param request checkout request
     * @return checkout response
     */
    @PostMapping
    @Operation(summary = "Initiate checkout", description = "Starts the checkout saga workflow")
    @ApiResponses(value = {
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "201", description = "Checkout initiated"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "400", description = "Invalid request")
    })
    public ResponseEntity<ApiResponse<CheckoutResponse>> initiateCheckout(
            @Valid @RequestBody CheckoutRequest request) {
        log.info("Received checkout request: userId={}, cartId={}",
            request.getUserId(), request.getCartId());

        // Build saga payload from checkout request
        Map<String, Object> payload = buildCheckoutPayload(request);

        StartSagaRequest sagaRequest = StartSagaRequest.builder()
            .type(SagaType.CHECKOUT)
            .userId(request.getUserId())
            .entityId(request.getCartId())
            .entityType("CART")
            .payload(payload)
            .priority(1)  // Higher priority for checkout
            .timeoutMs(300000)
            .build();

        SagaResponse sagaResponse = sagaService.startSaga(sagaRequest);

        CheckoutResponse response = CheckoutResponse.builder()
            .sagaId(sagaResponse.getId())
            .status(sagaResponse.getStatus().name())
            .createdAt(Instant.now())
            .checkoutData(Map.of(
                "correlationId", sagaResponse.getCorrelationId(),
                "totalSteps", sagaResponse.getTotalSteps()
            ))
            .build();

        return ResponseEntity.status(HttpStatus.CREATED)
            .body(ApiResponse.success(response));
    }

    /**
     * Gets checkout status.
     *
     * @param sagaId saga ID
     * @return checkout status
     */
    @GetMapping("/{sagaId}/status")
    @Operation(summary = "Get checkout status", description = "Retrieves checkout saga status")
    public ResponseEntity<ApiResponse<CheckoutResponse>> getCheckoutStatus(
            @PathVariable String sagaId) {
        log.debug("Getting checkout status: sagaId={}", sagaId);

        SagaResponse saga = sagaService.getSaga(sagaId);

        CheckoutResponse response = CheckoutResponse.builder()
            .sagaId(saga.getId())
            .status(saga.getStatus().name())
            .build();

        // Extract order and payment info from saga context
        if (saga.getContext() != null) {
            if (saga.getContext().containsKey("orderId")) {
                response.setOrderId((String) saga.getContext().get("orderId"));
            }
            if (saga.getContext().containsKey("paymentId")) {
                response.setPaymentId((String) saga.getContext().get("paymentId"));
            }
            if (saga.getContext().containsKey("paymentUrl")) {
                response.setPaymentUrl((String) saga.getContext().get("paymentUrl"));
            }
        }

        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Confirms payment for checkout.
     *
     * @param sagaId saga ID
     * @param paymentConfirmation payment confirmation data
     * @return updated checkout response
     */
    @PostMapping("/{sagaId}/confirm-payment")
    @Operation(summary = "Confirm payment", description = "Confirms payment for checkout")
    public ResponseEntity<ApiResponse<CheckoutResponse>> confirmPayment(
            @PathVariable String sagaId,
            @RequestBody Map<String, Object> paymentConfirmation) {
        log.info("Confirming payment: sagaId={}", sagaId);

        // Handle payment confirmation callback
        com.foodai.orchestration.dto.request.SagaCallbackRequest callback =
            com.foodai.orchestration.dto.request.SagaCallbackRequest.builder()
                .sagaId(sagaId)
                .stepId((String) paymentConfirmation.get("stepId"))
                .success(Boolean.TRUE.equals(paymentConfirmation.get("success")))
                .outputData(paymentConfirmation)
                .build();

        SagaResponse saga = sagaService.handleCallback(callback);

        CheckoutResponse response = CheckoutResponse.builder()
            .sagaId(saga.getId())
            .status(saga.getStatus().name())
            .build();

        if (saga.getContext() != null) {
            response.setOrderId((String) saga.getContext().get("orderId"));
            response.setPaymentId((String) saga.getContext().get("paymentId"));
        }

        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Cancels checkout.
     *
     * @param sagaId saga ID
     * @return cancelled checkout response
     */
    @PostMapping("/{sagaId}/cancel")
    @Operation(summary = "Cancel checkout", description = "Cancels the checkout process")
    public ResponseEntity<ApiResponse<CheckoutResponse>> cancelCheckout(
            @PathVariable String sagaId) {
        log.info("Cancelling checkout: sagaId={}", sagaId);

        SagaResponse saga = sagaService.cancelSaga(sagaId);

        CheckoutResponse response = CheckoutResponse.builder()
            .sagaId(saga.getId())
            .status(saga.getStatus().name())
            .build();

        return ResponseEntity.ok(ApiResponse.success(response));
    }

    private Map<String, Object> buildCheckoutPayload(CheckoutRequest request) {
        Map<String, Object> payload = new HashMap<>();
        payload.put("cartId", request.getCartId());
        payload.put("userId", request.getUserId());
        payload.put("deliveryAddressId", request.getDeliveryAddressId());
        payload.put("paymentMethod", request.getPaymentDetails().getMethod());
        payload.put("paymentProvider", request.getPaymentDetails().getProvider());

        if (request.getCouponCode() != null) {
            payload.put("couponCode", request.getCouponCode());
        }
        if (request.getDeliveryInstructions() != null) {
            payload.put("deliveryInstructions", request.getDeliveryInstructions());
        }
        if (request.getScheduledDeliveryTime() != null) {
            payload.put("scheduledDeliveryTime", request.getScheduledDeliveryTime());
        }
        payload.put("contactlessDelivery", request.isContactlessDelivery());

        if (request.getTipAmount() != null) {
            payload.put("tipAmount", request.getTipAmount());
        }

        return payload;
    }
}
