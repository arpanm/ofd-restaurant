package com.foodai.orchestration.controller;

import com.foodai.orchestration.domain.model.SagaStatus;
import com.foodai.orchestration.domain.model.SagaType;
import com.foodai.orchestration.dto.request.SagaCallbackRequest;
import com.foodai.orchestration.dto.request.StartSagaRequest;
import com.foodai.orchestration.dto.response.ApiResponse;
import com.foodai.orchestration.dto.response.SagaResponse;
import com.foodai.orchestration.service.SagaService;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.Parameter;
import io.swagger.v3.oas.annotations.responses.ApiResponses;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.Map;

/**
 * REST controller for saga management.
 *
 * @author FoodAI Team
 */
@RestController
@RequestMapping("/api/v1/sagas")
@RequiredArgsConstructor
@Slf4j
@Tag(name = "Saga Management", description = "APIs for managing saga orchestration")
public class SagaController {

    private final SagaService sagaService;

    /**
     * Starts a new saga.
     *
     * @param request start saga request
     * @return created saga
     */
    @PostMapping
    @Operation(summary = "Start a new saga", description = "Initiates a new saga workflow")
    @ApiResponses(value = {
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "201", description = "Saga created"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "400", description = "Invalid request")
    })
    public ResponseEntity<ApiResponse<SagaResponse>> startSaga(
            @Valid @RequestBody StartSagaRequest request) {
        log.info("Received request to start saga: type={}", request.getType());
        SagaResponse response = sagaService.startSaga(request);
        return ResponseEntity.status(HttpStatus.CREATED)
            .body(ApiResponse.success(response));
    }

    /**
     * Gets a saga by ID.
     *
     * @param sagaId saga ID
     * @return saga details
     */
    @GetMapping("/{sagaId}")
    @Operation(summary = "Get saga by ID", description = "Retrieves saga details by ID")
    @ApiResponses(value = {
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "200", description = "Saga found"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "404", description = "Saga not found")
    })
    public ResponseEntity<ApiResponse<SagaResponse>> getSaga(
            @Parameter(description = "Saga ID") @PathVariable String sagaId) {
        log.debug("Received request to get saga: sagaId={}", sagaId);
        SagaResponse response = sagaService.getSaga(sagaId);
        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Gets a saga by correlation ID.
     *
     * @param correlationId correlation ID
     * @return saga details
     */
    @GetMapping("/correlation/{correlationId}")
    @Operation(summary = "Get saga by correlation ID", description = "Retrieves saga by correlation ID")
    public ResponseEntity<ApiResponse<SagaResponse>> getSagaByCorrelationId(
            @Parameter(description = "Correlation ID") @PathVariable String correlationId) {
        log.debug("Received request to get saga by correlationId: {}", correlationId);
        SagaResponse response = sagaService.getSagaByCorrelationId(correlationId);
        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Gets sagas by user ID.
     *
     * @param userId user ID
     * @param page page number
     * @param size page size
     * @return page of sagas
     */
    @GetMapping("/user/{userId}")
    @Operation(summary = "Get sagas by user", description = "Retrieves sagas for a user")
    public ResponseEntity<ApiResponse<Page<SagaResponse>>> getSagasByUser(
            @Parameter(description = "User ID") @PathVariable String userId,
            @RequestParam(defaultValue = "0") int page,
            @RequestParam(defaultValue = "20") int size) {
        log.debug("Received request to get sagas by user: userId={}", userId);
        Pageable pageable = PageRequest.of(page, size, Sort.by(Sort.Direction.DESC, "createdAt"));
        Page<SagaResponse> response = sagaService.getSagasByUserId(userId, pageable);
        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Gets sagas by type and status.
     *
     * @param type saga type
     * @param status saga status
     * @param page page number
     * @param size page size
     * @return page of sagas
     */
    @GetMapping("/filter")
    @Operation(summary = "Filter sagas", description = "Retrieves sagas by type and status")
    public ResponseEntity<ApiResponse<Page<SagaResponse>>> filterSagas(
            @Parameter(description = "Saga type") @RequestParam SagaType type,
            @Parameter(description = "Saga status") @RequestParam SagaStatus status,
            @RequestParam(defaultValue = "0") int page,
            @RequestParam(defaultValue = "20") int size) {
        log.debug("Received request to filter sagas: type={}, status={}", type, status);
        Pageable pageable = PageRequest.of(page, size, Sort.by(Sort.Direction.DESC, "createdAt"));
        Page<SagaResponse> response = sagaService.getSagasByTypeAndStatus(type, status, pageable);
        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Handles saga step callback.
     *
     * @param request callback request
     * @return updated saga
     */
    @PostMapping("/callback")
    @Operation(summary = "Handle step callback", description = "Processes callback from a saga step")
    public ResponseEntity<ApiResponse<SagaResponse>> handleCallback(
            @Valid @RequestBody SagaCallbackRequest request) {
        log.info("Received saga callback: sagaId={}, stepId={}, success={}",
            request.getSagaId(), request.getStepId(), request.isSuccess());
        SagaResponse response = sagaService.handleCallback(request);
        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Cancels a saga.
     *
     * @param sagaId saga ID
     * @return cancelled saga
     */
    @PostMapping("/{sagaId}/cancel")
    @Operation(summary = "Cancel saga", description = "Cancels a running saga")
    @ApiResponses(value = {
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "200", description = "Saga cancelled"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "404", description = "Saga not found"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "409", description = "Cannot cancel saga")
    })
    public ResponseEntity<ApiResponse<SagaResponse>> cancelSaga(
            @Parameter(description = "Saga ID") @PathVariable String sagaId) {
        log.info("Received request to cancel saga: sagaId={}", sagaId);
        SagaResponse response = sagaService.cancelSaga(sagaId);
        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Retries a failed saga.
     *
     * @param sagaId saga ID
     * @return retried saga
     */
    @PostMapping("/{sagaId}/retry")
    @Operation(summary = "Retry saga", description = "Retries a failed saga")
    @ApiResponses(value = {
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "200", description = "Saga retry initiated"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "404", description = "Saga not found"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "409", description = "Cannot retry saga")
    })
    public ResponseEntity<ApiResponse<SagaResponse>> retrySaga(
            @Parameter(description = "Saga ID") @PathVariable String sagaId) {
        log.info("Received request to retry saga: sagaId={}", sagaId);
        SagaResponse response = sagaService.retrySaga(sagaId);
        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Gets saga statistics.
     *
     * @param type optional saga type filter
     * @return statistics
     */
    @GetMapping("/statistics")
    @Operation(summary = "Get saga statistics", description = "Retrieves saga statistics")
    public ResponseEntity<ApiResponse<Map<String, Object>>> getStatistics(
            @Parameter(description = "Saga type filter") @RequestParam(required = false) SagaType type) {
        log.debug("Received request for saga statistics: type={}", type);
        Map<String, Object> stats = sagaService.getStatistics(type);
        return ResponseEntity.ok(ApiResponse.success(stats));
    }
}
