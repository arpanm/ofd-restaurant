package com.foodai.orchestration.domain.model;

import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;
import org.springframework.data.annotation.CreatedDate;
import org.springframework.data.annotation.Id;
import org.springframework.data.annotation.LastModifiedDate;
import org.springframework.data.annotation.Version;
import org.springframework.data.mongodb.core.index.CompoundIndex;
import org.springframework.data.mongodb.core.index.CompoundIndexes;
import org.springframework.data.mongodb.core.index.Indexed;
import org.springframework.data.mongodb.core.mapping.Document;

import java.time.Instant;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.Optional;

/**
 * Aggregate root representing a saga execution.
 *
 * <p>A saga is a sequence of local transactions where each transaction
 * updates data within a single service. If a step fails, compensation
 * transactions are executed to undo the changes made by preceding steps.
 *
 * @author FoodAI Team
 */
@Document(collection = "sagas")
@CompoundIndexes({
    @CompoundIndex(name = "type_status_idx", def = "{'type': 1, 'status': 1}"),
    @CompoundIndex(name = "correlationId_idx", def = "{'correlationId': 1}"),
    @CompoundIndex(name = "userId_type_idx", def = "{'userId': 1, 'type': 1}")
})
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class Saga {

    @Id
    private String id;

    /** Type of saga */
    @Indexed
    private SagaType type;

    /** Current status of the saga */
    @Indexed
    private SagaStatus status;

    /** Correlation ID for tracing */
    @Indexed
    private String correlationId;

    /** User who initiated the saga */
    @Indexed
    private String userId;

    /** Reference to the primary entity (e.g., orderId, paymentId) */
    private String entityId;

    /** Entity type (e.g., "ORDER", "PAYMENT") */
    private String entityType;

    /** Saga definition name */
    private String sagaDefinition;

    /** Input payload for the saga */
    private Map<String, Object> payload;

    /** Saga context/state data */
    private Map<String, Object> context;

    /** List of saga steps */
    @Builder.Default
    private List<SagaStep> steps = new ArrayList<>();

    /** Index of current step being executed */
    private int currentStepIndex;

    /** Total number of steps */
    private int totalSteps;

    /** Number of completed steps */
    private int completedSteps;

    /** When the saga started */
    private Instant startedAt;

    /** When the saga completed */
    private Instant completedAt;

    /** Total duration in milliseconds */
    private Long totalDurationMs;

    /** Error message if saga failed */
    private String errorMessage;

    /** Error code if saga failed */
    private String errorCode;

    /** Number of retry attempts at saga level */
    private int retryCount;

    /** Maximum retries at saga level */
    @Builder.Default
    private int maxRetries = 3;

    /** Timeout for entire saga in milliseconds */
    @Builder.Default
    private long timeoutMs = 300000; // 5 minutes default

    /** Priority of this saga (higher = more urgent) */
    @Builder.Default
    private int priority = 0;

    /** Metadata for additional info */
    private Map<String, String> metadata;

    /** Version for optimistic locking */
    @Version
    private Long version;

    @CreatedDate
    private Instant createdAt;

    @LastModifiedDate
    private Instant updatedAt;

    /**
     * Starts the saga execution.
     */
    public void start() {
        if (this.status != SagaStatus.PENDING) {
            throw new IllegalStateException("Can only start a saga in PENDING status");
        }
        this.status = SagaStatus.RUNNING;
        this.startedAt = Instant.now();
        this.currentStepIndex = 0;
        this.totalSteps = steps.size();
    }

    /**
     * Adds a step to the saga.
     *
     * @param step the step to add
     */
    public void addStep(SagaStep step) {
        step.setStepOrder(steps.size());
        steps.add(step);
        this.totalSteps = steps.size();
    }

    /**
     * Gets the current step to execute.
     *
     * @return Optional containing current step or empty if all steps completed
     */
    public Optional<SagaStep> getCurrentStep() {
        if (currentStepIndex >= 0 && currentStepIndex < steps.size()) {
            return Optional.of(steps.get(currentStepIndex));
        }
        return Optional.empty();
    }

    /**
     * Advances to the next step.
     *
     * @return true if there is a next step, false if saga is complete
     */
    public boolean advanceToNextStep() {
        if (currentStepIndex < steps.size() - 1) {
            currentStepIndex++;
            return true;
        }
        return false;
    }

    /**
     * Marks the current step as completed and advances.
     *
     * @param output output data from the step
     */
    public void completeCurrentStep(Map<String, Object> output) {
        getCurrentStep().ifPresent(step -> {
            step.complete(output);
            completedSteps++;
            if (context != null && output != null) {
                context.putAll(output);
            }
        });

        if (!advanceToNextStep()) {
            complete();
        }
    }

    /**
     * Marks the current step as failed and initiates compensation.
     *
     * @param errorCode error code
     * @param errorMessage error message
     */
    public void failCurrentStep(String errorCode, String errorMessage) {
        getCurrentStep().ifPresent(step -> step.fail(errorCode, errorMessage));
        this.errorCode = errorCode;
        this.errorMessage = errorMessage;
        startCompensation();
    }

    /**
     * Completes the saga successfully.
     */
    public void complete() {
        this.status = SagaStatus.COMPLETED;
        this.completedAt = Instant.now();
        calculateTotalDuration();
    }

    /**
     * Starts the compensation process (rollback).
     */
    public void startCompensation() {
        this.status = SagaStatus.COMPENSATING;
        // Start from the last completed step
        this.currentStepIndex = completedSteps - 1;
    }

    /**
     * Compensates the current step and moves to previous.
     */
    public void compensateCurrentStep() {
        getCurrentStep().ifPresent(step -> {
            if (step.needsCompensation()) {
                step.completeCompensation();
            }
        });

        if (currentStepIndex > 0) {
            currentStepIndex--;
        } else {
            completeCompensation();
        }
    }

    /**
     * Marks compensation as complete.
     */
    public void completeCompensation() {
        this.status = SagaStatus.COMPENSATED;
        this.completedAt = Instant.now();
        calculateTotalDuration();
    }

    /**
     * Marks the saga as failed (unrecoverable).
     *
     * @param errorCode error code
     * @param errorMessage error message
     */
    public void fail(String errorCode, String errorMessage) {
        this.status = SagaStatus.FAILED;
        this.errorCode = errorCode;
        this.errorMessage = errorMessage;
        this.completedAt = Instant.now();
        calculateTotalDuration();
    }

    /**
     * Cancels the saga.
     */
    public void cancel() {
        if (status == SagaStatus.RUNNING || status == SagaStatus.WAITING) {
            this.status = SagaStatus.CANCELLED;
            this.completedAt = Instant.now();
            calculateTotalDuration();
        }
    }

    /**
     * Pauses the saga for manual intervention.
     */
    public void pause() {
        if (status == SagaStatus.RUNNING) {
            this.status = SagaStatus.PAUSED;
        }
    }

    /**
     * Resumes a paused saga.
     */
    public void resume() {
        if (status == SagaStatus.PAUSED) {
            this.status = SagaStatus.RUNNING;
        }
    }

    /**
     * Sets the saga to waiting state for external callback.
     */
    public void waitForCallback() {
        this.status = SagaStatus.WAITING;
    }

    /**
     * Checks if the saga is in a terminal state.
     *
     * @return true if saga is in terminal state
     */
    public boolean isTerminal() {
        return status == SagaStatus.COMPLETED
            || status == SagaStatus.COMPENSATED
            || status == SagaStatus.FAILED
            || status == SagaStatus.CANCELLED;
    }

    /**
     * Checks if the saga can be retried.
     *
     * @return true if retry is possible
     */
    public boolean canRetry() {
        return retryCount < maxRetries
            && (status == SagaStatus.FAILED || status == SagaStatus.COMPENSATED);
    }

    /**
     * Retries the saga from the beginning.
     *
     * @return true if retry was initiated
     */
    public boolean retry() {
        if (canRetry()) {
            retryCount++;
            this.status = SagaStatus.PENDING;
            this.currentStepIndex = 0;
            this.completedSteps = 0;
            this.errorCode = null;
            this.errorMessage = null;
            // Reset all steps
            steps.forEach(step -> {
                step.setStatus(SagaStepStatus.PENDING);
                step.setErrorCode(null);
                step.setErrorMessage(null);
                step.setRetryCount(0);
                step.setCompensationExecuted(false);
            });
            return true;
        }
        return false;
    }

    /**
     * Gets the progress percentage.
     *
     * @return progress as percentage (0-100)
     */
    public int getProgressPercent() {
        if (totalSteps == 0) {
            return 0;
        }
        return (completedSteps * 100) / totalSteps;
    }

    /**
     * Validates the saga state.
     *
     * @throws IllegalStateException if saga is invalid
     */
    public void validate() {
        if (type == null) {
            throw new IllegalStateException("Saga type is required");
        }
        if (correlationId == null || correlationId.isBlank()) {
            throw new IllegalStateException("Correlation ID is required");
        }
        if (steps.isEmpty()) {
            throw new IllegalStateException("Saga must have at least one step");
        }
    }

    private void calculateTotalDuration() {
        if (startedAt != null && completedAt != null) {
            this.totalDurationMs = completedAt.toEpochMilli() - startedAt.toEpochMilli();
        }
    }

    /**
     * Updates the saga context with new data.
     *
     * @param key context key
     * @param value context value
     */
    public void updateContext(String key, Object value) {
        if (this.context == null) {
            this.context = new java.util.HashMap<>();
        }
        this.context.put(key, value);
    }
}
