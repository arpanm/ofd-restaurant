package com.foodai.orchestration.domain.model;

/**
 * Status of a saga execution.
 *
 * @author FoodAI Team
 */
public enum SagaStatus {
    /** Saga is created but not yet started */
    PENDING,
    /** Saga is currently executing */
    RUNNING,
    /** Saga completed successfully */
    COMPLETED,
    /** Saga failed and compensation is in progress */
    COMPENSATING,
    /** Saga failed and all compensations completed */
    COMPENSATED,
    /** Saga failed and compensation also failed */
    FAILED,
    /** Saga was manually cancelled */
    CANCELLED,
    /** Saga is waiting for external event */
    WAITING,
    /** Saga is paused for manual intervention */
    PAUSED
}
