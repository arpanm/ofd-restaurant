package com.foodai.orchestration.domain.model;

import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

import java.time.Instant;
import java.util.Map;

/**
 * Represents a single step in a saga execution.
 *
 * <p>Each step corresponds to an action performed on a microservice,
 * with associated compensation logic for rollback scenarios.
 *
 * @author FoodAI Team
 */
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class SagaStep {

    /** Unique identifier for this step within the saga */
    private String stepId;

    /** Name of the step (e.g., "validateCart", "processPayment") */
    private String stepName;

    /** Order of execution (0-indexed) */
    private int stepOrder;

    /** Target service for this step */
    private String targetService;

    /** Action to execute */
    private String action;

    /** Compensation action for rollback */
    private String compensationAction;

    /** Current status of this step */
    private SagaStepStatus status;

    /** Input data for this step */
    private Map<String, Object> inputData;

    /** Output data from this step */
    private Map<String, Object> outputData;

    /** Error message if step failed */
    private String errorMessage;

    /** Error code if step failed */
    private String errorCode;

    /** Number of retry attempts */
    private int retryCount;

    /** Maximum retry attempts allowed */
    private int maxRetries;

    /** Timeout in milliseconds */
    private long timeoutMs;

    /** When the step started executing */
    private Instant startedAt;

    /** When the step completed */
    private Instant completedAt;

    /** Duration in milliseconds */
    private Long durationMs;

    /** Whether this step is compensatable */
    private boolean compensatable;

    /** Whether this step executes asynchronously (waits for callback) */
    private boolean async;

    /** Whether compensation was executed */
    private boolean compensationExecuted;

    /** Compensation error if failed */
    private String compensationError;

    /**
     * Starts the step execution.
     */
    public void start() {
        this.status = SagaStepStatus.EXECUTING;
        this.startedAt = Instant.now();
    }

    /**
     * Marks the step as completed successfully.
     *
     * @param output output data from execution
     */
    public void complete(Map<String, Object> output) {
        this.status = SagaStepStatus.COMPLETED;
        this.outputData = output;
        this.completedAt = Instant.now();
        calculateDuration();
    }

    /**
     * Marks the step as failed.
     *
     * @param errorCode error code
     * @param errorMessage error message
     */
    public void fail(String errorCode, String errorMessage) {
        this.status = SagaStepStatus.FAILED;
        this.errorCode = errorCode;
        this.errorMessage = errorMessage;
        this.completedAt = Instant.now();
        calculateDuration();
    }

    /**
     * Increments the retry count.
     *
     * @return true if retry is allowed, false if max retries exceeded
     */
    public boolean retry() {
        if (retryCount < maxRetries) {
            retryCount++;
            this.status = SagaStepStatus.PENDING;
            return true;
        }
        return false;
    }

    /**
     * Starts compensation for this step.
     */
    public void startCompensation() {
        if (compensatable && status == SagaStepStatus.COMPLETED) {
            this.status = SagaStepStatus.COMPENSATING;
        }
    }

    /**
     * Marks compensation as completed.
     */
    public void completeCompensation() {
        this.status = SagaStepStatus.COMPENSATED;
        this.compensationExecuted = true;
    }

    /**
     * Marks compensation as failed.
     *
     * @param error compensation error message
     */
    public void failCompensation(String error) {
        this.status = SagaStepStatus.COMPENSATION_FAILED;
        this.compensationError = error;
        this.compensationExecuted = true;
    }

    /**
     * Checks if the step can be retried.
     *
     * @return true if retry is possible
     */
    public boolean canRetry() {
        return retryCount < maxRetries && status == SagaStepStatus.FAILED;
    }

    /**
     * Checks if the step needs compensation.
     *
     * @return true if compensation is needed
     */
    public boolean needsCompensation() {
        return compensatable && status == SagaStepStatus.COMPLETED && !compensationExecuted;
    }

    private void calculateDuration() {
        if (startedAt != null && completedAt != null) {
            this.durationMs = completedAt.toEpochMilli() - startedAt.toEpochMilli();
        }
    }
}
