package com.foodai.orchestration.domain.model;

/**
 * Types of sagas supported by the orchestration service.
 *
 * @author FoodAI Team
 */
public enum SagaType {
    /** Standard order creation saga */
    ORDER_CREATE,
    /** Multi-restaurant order saga */
    MULTI_RESTAURANT_ORDER,
    /** Order cancellation saga */
    ORDER_CANCEL,
    /** Checkout flow saga */
    CHECKOUT,
    /** Payment processing saga */
    PAYMENT,
    /** Refund processing saga */
    REFUND,
    /** Rider assignment saga */
    RIDER_ASSIGNMENT,
    /** Diet plan order saga */
    DIET_PLAN_ORDER,
    /** Party planning order saga */
    PARTY_PLAN_ORDER,
    /** AI chat based order saga */
    AI_CHAT_ORDER,
    /** User registration saga */
    USER_REGISTRATION,
    /** Restaurant onboarding saga */
    RESTAURANT_ONBOARDING,
    /** Subscription renewal saga */
    SUBSCRIPTION_RENEWAL,
    /** Loyalty points redemption saga */
    LOYALTY_REDEMPTION,
    /** Promotion application saga */
    PROMOTION_APPLICATION
}
