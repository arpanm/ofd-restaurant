package com.foodai.orchestration.infrastructure.messaging;

import com.foodai.orchestration.saga.executor.SagaExecutor;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.kafka.annotation.KafkaListener;
import org.springframework.messaging.handler.annotation.Payload;
import org.springframework.stereotype.Component;

import java.util.Map;

import static net.logstash.logback.argument.StructuredArguments.kv;

/**
 * Consumer for saga-related events from Kafka.
 *
 * @author FoodAI Team
 */
@Component
@RequiredArgsConstructor
@Slf4j
public class SagaEventConsumer {

    private final SagaExecutor sagaExecutor;

    /**
     * Handles step completion callbacks from services.
     *
     * @param payload callback payload
     */
    @KafkaListener(
        topics = "orchestration.step.callback",
        groupId = "orchestration-service",
        containerFactory = "kafkaListenerContainerFactory"
    )
    public void handleStepCallback(@Payload Map<String, Object> payload) {
        String sagaId = (String) payload.get("sagaId");
        String stepId = (String) payload.get("stepId");
        boolean success = Boolean.TRUE.equals(payload.get("success"));

        log.info("Received step callback",
            kv("sagaId", sagaId),
            kv("stepId", stepId),
            kv("success", success));

        try {
            @SuppressWarnings("unchecked")
            Map<String, Object> output = (Map<String, Object>) payload.get("output");
            String errorCode = (String) payload.get("errorCode");
            String errorMessage = (String) payload.get("errorMessage");

            sagaExecutor.handleStepResult(sagaId, stepId, success, output, errorCode, errorMessage);

            log.info("Step callback processed successfully",
                kv("sagaId", sagaId),
                kv("stepId", stepId));

        } catch (Exception e) {
            log.error("Failed to process step callback",
                kv("sagaId", sagaId),
                kv("stepId", stepId),
                e);
        }
    }

    /**
     * Handles payment completion events.
     *
     * @param payload payment event payload
     */
    @KafkaListener(
        topics = "payment.completed",
        groupId = "orchestration-service",
        containerFactory = "kafkaListenerContainerFactory"
    )
    public void handlePaymentCompleted(@Payload Map<String, Object> payload) {
        String sagaId = (String) payload.get("sagaId");
        String paymentId = (String) payload.get("paymentId");
        String status = (String) payload.get("status");

        log.info("Received payment completed event",
            kv("sagaId", sagaId),
            kv("paymentId", paymentId),
            kv("status", status));

        if (sagaId != null) {
            try {
                boolean success = "COMPLETED".equals(status) || "SUCCESS".equals(status);
                Map<String, Object> output = Map.of(
                    "paymentId", paymentId != null ? paymentId : "",
                    "status", status != null ? status : "",
                    "amount", payload.getOrDefault("amount", 0)
                );

                String stepId = (String) payload.get("stepId");
                if (stepId != null) {
                    sagaExecutor.handleStepResult(sagaId, stepId, success, output,
                        success ? null : "PAYMENT_FAILED",
                        success ? null : "Payment processing failed");
                }
            } catch (Exception e) {
                log.error("Failed to process payment completed event",
                    kv("sagaId", sagaId),
                    e);
            }
        }
    }

    /**
     * Handles order status change events.
     *
     * @param payload order event payload
     */
    @KafkaListener(
        topics = "order.status.changed",
        groupId = "orchestration-service",
        containerFactory = "kafkaListenerContainerFactory"
    )
    public void handleOrderStatusChanged(@Payload Map<String, Object> payload) {
        String orderId = (String) payload.get("orderId");
        String status = (String) payload.get("status");
        String sagaId = (String) payload.get("sagaId");

        log.info("Received order status changed event",
            kv("orderId", orderId),
            kv("status", status),
            kv("sagaId", sagaId));

        // Handle order status changes that might affect sagas
        // For example, if order is confirmed, continue saga
        // If order is cancelled externally, compensate saga
    }

    /**
     * Handles rider assignment events.
     *
     * @param payload rider event payload
     */
    @KafkaListener(
        topics = "rider.assigned",
        groupId = "orchestration-service",
        containerFactory = "kafkaListenerContainerFactory"
    )
    public void handleRiderAssigned(@Payload Map<String, Object> payload) {
        String orderId = (String) payload.get("orderId");
        String riderId = (String) payload.get("riderId");
        String sagaId = (String) payload.get("sagaId");

        log.info("Received rider assigned event",
            kv("orderId", orderId),
            kv("riderId", riderId),
            kv("sagaId", sagaId));

        if (sagaId != null) {
            try {
                String stepId = (String) payload.get("stepId");
                if (stepId != null) {
                    Map<String, Object> output = Map.of(
                        "riderId", riderId != null ? riderId : "",
                        "orderId", orderId != null ? orderId : ""
                    );
                    sagaExecutor.handleStepResult(sagaId, stepId, true, output, null, null);
                }
            } catch (Exception e) {
                log.error("Failed to process rider assigned event",
                    kv("sagaId", sagaId),
                    e);
            }
        }
    }
}
