package com.foodai.orchestration.infrastructure.messaging;

import com.foodai.orchestration.domain.model.Saga;
import com.foodai.orchestration.domain.model.SagaStep;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.stereotype.Component;

import java.time.Instant;
import java.util.HashMap;
import java.util.Map;
import java.util.UUID;

import static net.logstash.logback.argument.StructuredArguments.kv;

/**
 * Publisher for saga-related events to Kafka.
 *
 * @author FoodAI Team
 */
@Component
@RequiredArgsConstructor
@Slf4j
public class SagaEventPublisher {

    private final KafkaTemplate<String, Object> kafkaTemplate;

    private static final String SAGA_EVENTS_TOPIC = "orchestration.saga.events";
    private static final String STEP_EXECUTION_TOPIC = "orchestration.step.execute";
    private static final String STEP_COMPENSATION_TOPIC = "orchestration.step.compensate";

    /**
     * Publishes a saga started event.
     *
     * @param saga the saga
     */
    public void publishSagaStarted(Saga saga) {
        Map<String, Object> event = createSagaEvent(saga, "SAGA_STARTED");
        publishEvent(SAGA_EVENTS_TOPIC, saga.getId(), event);
        log.info("Published saga started event",
            kv("sagaId", saga.getId()),
            kv("type", saga.getType()));
    }

    /**
     * Publishes a saga completed event.
     *
     * @param saga the saga
     */
    public void publishSagaCompleted(Saga saga) {
        Map<String, Object> event = createSagaEvent(saga, "SAGA_COMPLETED");
        event.put("totalDurationMs", saga.getTotalDurationMs());
        publishEvent(SAGA_EVENTS_TOPIC, saga.getId(), event);
        log.info("Published saga completed event",
            kv("sagaId", saga.getId()),
            kv("type", saga.getType()),
            kv("durationMs", saga.getTotalDurationMs()));
    }

    /**
     * Publishes a saga failed event.
     *
     * @param saga the saga
     */
    public void publishSagaFailed(Saga saga) {
        Map<String, Object> event = createSagaEvent(saga, "SAGA_FAILED");
        event.put("errorCode", saga.getErrorCode());
        event.put("errorMessage", saga.getErrorMessage());
        publishEvent(SAGA_EVENTS_TOPIC, saga.getId(), event);
        log.info("Published saga failed event",
            kv("sagaId", saga.getId()),
            kv("type", saga.getType()),
            kv("errorCode", saga.getErrorCode()));
    }

    /**
     * Publishes a saga compensated event.
     *
     * @param saga the saga
     */
    public void publishSagaCompensated(Saga saga) {
        Map<String, Object> event = createSagaEvent(saga, "SAGA_COMPENSATED");
        publishEvent(SAGA_EVENTS_TOPIC, saga.getId(), event);
        log.info("Published saga compensated event",
            kv("sagaId", saga.getId()),
            kv("type", saga.getType()));
    }

    /**
     * Publishes a step execution command.
     *
     * @param saga the saga
     * @param step the step
     * @param input input data for the step
     */
    public void publishStepExecution(Saga saga, SagaStep step, Map<String, Object> input) {
        String topic = getStepTopic(step.getTargetService());

        Map<String, Object> command = new HashMap<>();
        command.put("id", UUID.randomUUID().toString());
        command.put("type", "STEP_EXECUTE");
        command.put("sagaId", saga.getId());
        command.put("stepId", step.getStepId());
        command.put("correlationId", saga.getCorrelationId());
        command.put("action", step.getAction());
        command.put("targetService", step.getTargetService());
        command.put("timestamp", Instant.now().toString());
        command.put("input", input);
        command.put("metadata", Map.of(
            "userId", saga.getUserId() != null ? saga.getUserId() : "",
            "entityId", saga.getEntityId() != null ? saga.getEntityId() : "",
            "sagaType", saga.getType().name()
        ));

        publishEvent(topic, saga.getCorrelationId(), command);
        log.info("Published step execution command",
            kv("sagaId", saga.getId()),
            kv("stepId", step.getStepId()),
            kv("action", step.getAction()),
            kv("topic", topic));
    }

    /**
     * Publishes a step compensation command.
     *
     * @param saga the saga
     * @param step the step
     */
    public void publishStepCompensation(Saga saga, SagaStep step) {
        String topic = getStepTopic(step.getTargetService());

        Map<String, Object> command = new HashMap<>();
        command.put("id", UUID.randomUUID().toString());
        command.put("type", "STEP_COMPENSATE");
        command.put("sagaId", saga.getId());
        command.put("stepId", step.getStepId());
        command.put("correlationId", saga.getCorrelationId());
        command.put("action", step.getCompensationAction());
        command.put("targetService", step.getTargetService());
        command.put("timestamp", Instant.now().toString());
        command.put("originalOutput", step.getOutputData());
        command.put("metadata", Map.of(
            "userId", saga.getUserId() != null ? saga.getUserId() : "",
            "entityId", saga.getEntityId() != null ? saga.getEntityId() : "",
            "sagaType", saga.getType().name()
        ));

        publishEvent(topic, saga.getCorrelationId(), command);
        log.info("Published step compensation command",
            kv("sagaId", saga.getId()),
            kv("stepId", step.getStepId()),
            kv("compensationAction", step.getCompensationAction()),
            kv("topic", topic));
    }

    private Map<String, Object> createSagaEvent(Saga saga, String eventType) {
        Map<String, Object> event = new HashMap<>();
        event.put("id", UUID.randomUUID().toString());
        event.put("type", eventType);
        event.put("sagaId", saga.getId());
        event.put("sagaType", saga.getType().name());
        event.put("correlationId", saga.getCorrelationId());
        event.put("userId", saga.getUserId());
        event.put("entityId", saga.getEntityId());
        event.put("entityType", saga.getEntityType());
        event.put("status", saga.getStatus().name());
        event.put("timestamp", Instant.now().toString());
        event.put("source", "orchestration-service");
        return event;
    }

    private String getStepTopic(String targetService) {
        // Generate topic name based on target service
        return "orchestration." + targetService.replace("-", ".") + ".commands";
    }

    private void publishEvent(String topic, String key, Object event) {
        try {
            kafkaTemplate.send(topic, key, event);
        } catch (Exception e) {
            log.error("Failed to publish event",
                kv("topic", topic),
                kv("key", key),
                e);
        }
    }
}
