package com.foodai.orchestration.saga.definition;

import com.foodai.orchestration.domain.model.SagaType;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

import java.util.ArrayList;
import java.util.List;

/**
 * Definition of a complete saga workflow.
 *
 * @author FoodAI Team
 */
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class SagaDefinition {

    /** Unique name of the saga definition */
    private String name;

    /** Saga type */
    private SagaType type;

    /** Description of what this saga does */
    private String description;

    /** Ordered list of steps */
    @Builder.Default
    private List<SagaStepDefinition> steps = new ArrayList<>();

    /** Default timeout for entire saga in milliseconds */
    @Builder.Default
    private long defaultTimeoutMs = 300000; // 5 minutes

    /** Maximum retries at saga level */
    @Builder.Default
    private int maxRetries = 3;

    /** Version of this definition */
    @Builder.Default
    private String version = "1.0";

    /**
     * Adds a step to the saga definition.
     *
     * @param step step definition
     * @return this definition for chaining
     */
    public SagaDefinition addStep(SagaStepDefinition step) {
        this.steps.add(step);
        return this;
    }

    /**
     * Creates a builder with a step.
     *
     * @param name step name
     * @param service target service
     * @param action action to execute
     * @return step builder
     */
    public static SagaStepDefinition.SagaStepDefinitionBuilder step(
            String name, String service, String action) {
        return SagaStepDefinition.builder()
            .name(name)
            .targetService(service)
            .action(action);
    }
}
