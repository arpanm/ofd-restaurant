package com.foodai.orchestration.saga.definition;

import com.foodai.orchestration.domain.model.SagaType;
import jakarta.annotation.PostConstruct;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.util.EnumMap;
import java.util.Map;
import java.util.Optional;

/**
 * Registry for saga definitions.
 *
 * <p>Contains all saga workflow definitions for the platform.
 *
 * @author FoodAI Team
 */
@Component
@Slf4j
public class SagaDefinitionRegistry {

    private final Map<SagaType, SagaDefinition> definitions = new EnumMap<>(SagaType.class);

    @PostConstruct
    public void init() {
        registerOrderCreateSaga();
        registerCheckoutSaga();
        registerRefundSaga();
        registerMultiRestaurantOrderSaga();
        registerRiderAssignmentSaga();
        registerDietPlanOrderSaga();
        registerPartyPlanOrderSaga();
        registerUserRegistrationSaga();
        log.info("Registered {} saga definitions", definitions.size());
    }

    /**
     * Gets a saga definition by type.
     *
     * @param type saga type
     * @return Optional containing definition
     */
    public Optional<SagaDefinition> getDefinition(SagaType type) {
        return Optional.ofNullable(definitions.get(type));
    }

    private void registerOrderCreateSaga() {
        SagaDefinition saga = SagaDefinition.builder()
            .name("order-create-saga")
            .type(SagaType.ORDER_CREATE)
            .description("Creates a new order from cart")
            .defaultTimeoutMs(300000)
            .build();

        saga.addStep(SagaStepDefinition.builder()
            .name("validateCart")
            .targetService("order-service")
            .action("VALIDATE_CART")
            .compensatable(false)
            .timeoutMs(10000)
            .build());

        saga.addStep(SagaStepDefinition.builder()
            .name("validateInventory")
            .targetService("menu-service")
            .action("VALIDATE_INVENTORY")
            .compensatable(false)
            .timeoutMs(15000)
            .build());

        saga.addStep(SagaStepDefinition.builder()
            .name("reserveInventory")
            .targetService("menu-service")
            .action("RESERVE_INVENTORY")
            .compensationAction("RELEASE_INVENTORY")
            .compensatable(true)
            .timeoutMs(15000)
            .build());

        saga.addStep(SagaStepDefinition.builder()
            .name("processPayment")
            .targetService("payment-service")
            .action("PROCESS_PAYMENT")
            .compensationAction("REFUND_PAYMENT")
            .compensatable(true)
            .async(true)
            .timeoutMs(60000)
            .build());

        saga.addStep(SagaStepDefinition.builder()
            .name("createOrder")
            .targetService("order-service")
            .action("CREATE_ORDER")
            .compensationAction("CANCEL_ORDER")
            .compensatable(true)
            .timeoutMs(15000)
            .build());

        saga.addStep(SagaStepDefinition.builder()
            .name("notifyRestaurant")
            .targetService("notification-service")
            .action("NOTIFY_RESTAURANT")
            .compensatable(false)
            .timeoutMs(10000)
            .build());

        saga.addStep(SagaStepDefinition.builder()
            .name("clearCart")
            .targetService("order-service")
            .action("CLEAR_CART")
            .compensatable(false)
            .timeoutMs(10000)
            .build());

        definitions.put(SagaType.ORDER_CREATE, saga);
    }
