package com.foodai.orchestration.saga.definition;

import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

/**
 * Definition of a saga step.
 *
 * @author FoodAI Team
 */
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class SagaStepDefinition {

    /** Step name */
    private String name;

    /** Target service to execute */
    private String targetService;

    /** Action to execute */
    private String action;

    /** Compensation action for rollback */
    private String compensationAction;

    /** Whether this step is compensatable */
    @Builder.Default
    private boolean compensatable = true;

    /** Maximum retries for this step */
    @Builder.Default
    private int maxRetries = 3;

    /** Timeout in milliseconds */
    @Builder.Default
    private long timeoutMs = 30000;

    /** Whether to wait for callback (async step) */
    @Builder.Default
    private boolean async = false;

    /** Input mapping from saga context */
    private String inputMapping;

    /** Output mapping to saga context */
    private String outputMapping;
}
