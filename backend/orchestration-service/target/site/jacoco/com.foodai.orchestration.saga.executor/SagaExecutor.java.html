package com.foodai.orchestration.saga.executor;

import com.foodai.orchestration.domain.model.Saga;
import com.foodai.orchestration.domain.model.SagaStatus;
import com.foodai.orchestration.domain.model.SagaStep;
import com.foodai.orchestration.domain.model.SagaStepStatus;
import com.foodai.orchestration.domain.repository.SagaRepository;
import com.foodai.orchestration.exception.SagaExecutionException;
import com.foodai.orchestration.exception.SagaNotFoundException;
import com.foodai.orchestration.infrastructure.messaging.SagaEventPublisher;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Transactional;

import java.util.Map;

import static net.logstash.logback.argument.StructuredArguments.kv;

/**
 * Executes saga workflows step by step.
 *
 * @author FoodAI Team
 */
@Component
@RequiredArgsConstructor
@Slf4j
public class SagaExecutor {

    private final SagaRepository sagaRepository;
    private final SagaEventPublisher eventPublisher;

    /**
     * Executes a saga by ID.
     *
     * @param sagaId saga ID
     */
    @Transactional
    public void execute(String sagaId) {
        log.info("Executing saga", kv("sagaId", sagaId));

        Saga saga = sagaRepository.findById(sagaId)
            .orElseThrow(() -> new SagaNotFoundException(sagaId));

        if (saga.getStatus() == SagaStatus.PENDING) {
            saga.start();
            saga = sagaRepository.save(saga);
        }

        if (saga.getStatus() != SagaStatus.RUNNING) {
            log.warn("Saga not in running state",
                kv("sagaId", sagaId),
                kv("status", saga.getStatus()));
            return;
        }

        try {
            executeNextStep(saga);
        } catch (Exception e) {
            log.error("Saga execution failed",
                kv("sagaId", sagaId), e);
            handleSagaFailure(saga, e);
        }
    }

    private void executeNextStep(Saga saga) {
        SagaStep currentStep = saga.getCurrentStep().orElse(null);
        if (currentStep == null) {
            saga.complete();
            sagaRepository.save(saga);
            eventPublisher.publishSagaCompleted(saga);
            log.info("Saga completed", kv("sagaId", saga.getId()));
            return;
        }

        log.info("Executing saga step",
            kv("sagaId", saga.getId()),
            kv("stepName", currentStep.getStepName()),
            kv("stepOrder", currentStep.getStepOrder()));

        currentStep.start();
        saga = sagaRepository.save(saga);

        try {
            // Publish step execution event
            Map<String, Object> stepInput = prepareStepInput(saga, currentStep);
            eventPublisher.publishStepExecution(saga, currentStep, stepInput);

            if (currentStep.isAsync()) {
                // For async steps, wait for callback
                saga.waitForCallback();
                sagaRepository.save(saga);
                log.info("Step waiting for callback",
                    kv("sagaId", saga.getId()),
                    kv("stepId", currentStep.getStepId()));
            } else {
                // For sync steps, simulate completion
                // In real implementation, this would wait for service response
                simulateStepCompletion(saga, currentStep);
            }
        } catch (Exception e) {
            handleStepFailure(saga, currentStep, e);
        }
    }

    private Map<String, Object> prepareStepInput(Saga saga, SagaStep step) {
        Map<String, Object> input = new java.util.HashMap<>();
        input.put("sagaId", saga.getId());
        input.put("stepId", step.getStepId());
        input.put("correlationId", saga.getCorrelationId());
        input.put("userId", saga.getUserId());
        input.put("entityId", saga.getEntityId());

        // Add saga context
        if (saga.getContext() != null) {
            input.putAll(saga.getContext());
        }

        // Add step-specific input if available
        if (step.getInputData() != null) {
            input.putAll(step.getInputData());
        }

        return input;
    }

    private void simulateStepCompletion(Saga saga, SagaStep step) {
        // In production, this would receive response from target service
        Map<String, Object> output = Map.of(
            "status", "SUCCESS",
            "stepId", step.getStepId()
        );

        saga.completeCurrentStep(output);
        saga = sagaRepository.save(saga);

        log.info("Step completed",
            kv("sagaId", saga.getId()),
            kv("stepName", step.getStepName()));

        // Continue to next step
        if (saga.getStatus() == SagaStatus.RUNNING) {
            executeNextStep(saga);
        } else if (saga.getStatus() == SagaStatus.COMPLETED) {
            eventPublisher.publishSagaCompleted(saga);
        }
    }

    private void handleStepFailure(Saga saga, SagaStep step, Exception e) {
        String errorCode = "STEP_EXECUTION_FAILED";
        String errorMessage = e.getMessage();

        if (e instanceof SagaExecutionException) {
            errorCode = ((SagaExecutionException) e).getErrorCode();
        }

        log.error("Step execution failed",
            kv("sagaId", saga.getId()),
            kv("stepName", step.getStepName()),
            kv("errorCode", errorCode),
            e);

        // Try retry first
        if (step.canRetry()) {
            log.info("Retrying step",
                kv("sagaId", saga.getId()),
                kv("stepName", step.getStepName()),
                kv("retryCount", step.getRetryCount() + 1));
            step.retry();
            saga = sagaRepository.save(saga);
            executeNextStep(saga);
            return;
        }

        // Fail step and start compensation
        saga.failCurrentStep(errorCode, errorMessage);
        saga = sagaRepository.save(saga);

        if (saga.getStatus() == SagaStatus.COMPENSATING) {
            executeCompensation(saga);
        }
    }

    private void handleSagaFailure(Saga saga, Exception e) {
        String errorCode = "SAGA_EXECUTION_FAILED";
        String errorMessage = e.getMessage();

        if (e instanceof SagaExecutionException) {
            errorCode = ((SagaExecutionException) e).getErrorCode();
        }

        saga.fail(errorCode, errorMessage);
        sagaRepository.save(saga);
        eventPublisher.publishSagaFailed(saga);
    }

    /**
     * Executes compensation for failed saga.
     *
     * @param saga the saga
     */
    @Transactional
    public void executeCompensation(Saga saga) {
        log.info("Starting compensation",
            kv("sagaId", saga.getId()),
            kv("currentStep", saga.getCurrentStepIndex()));

        while (saga.getCurrentStepIndex() >= 0) {
            SagaStep step = saga.getCurrentStep().orElse(null);
            if (step == null) {
                break;
            }

            if (step.needsCompensation()) {
                try {
                    log.info("Compensating step",
                        kv("sagaId", saga.getId()),
                        kv("stepName", step.getStepName()));

                    step.startCompensation();
                    saga = sagaRepository.save(saga);

                    // Publish compensation event
                    eventPublisher.publishStepCompensation(saga, step);

                    // Simulate compensation completion
                    step.completeCompensation();
                    saga = sagaRepository.save(saga);

                    log.info("Step compensated",
                        kv("sagaId", saga.getId()),
                        kv("stepName", step.getStepName()));

                } catch (Exception e) {
                    log.error("Compensation failed",
                        kv("sagaId", saga.getId()),
                        kv("stepName", step.getStepName()),
                        e);
                    step.failCompensation(e.getMessage());
                    saga = sagaRepository.save(saga);
                }
            }

            saga.compensateCurrentStep();
            saga = sagaRepository.save(saga);
        }

        saga.completeCompensation();
        saga = sagaRepository.save(saga);
        eventPublisher.publishSagaCompensated(saga);

        log.info("Compensation completed", kv("sagaId", saga.getId()));
    }

    /**
     * Handles step completion from external service.
     *
     * @param sagaId saga ID
     * @param stepId step ID
     * @param success whether step succeeded
     * @param output output data
     * @param errorCode error code if failed
     * @param errorMessage error message if failed
     */
    @Transactional
    public void handleStepResult(
            String sagaId,
            String stepId,
            boolean success,
            Map<String, Object> output,
            String errorCode,
            String errorMessage) {

        log.info("Handling step result",
            kv("sagaId", sagaId),
            kv("stepId", stepId),
            kv("success", success));

        Saga saga = sagaRepository.findById(sagaId)
            .orElseThrow(() -> new SagaNotFoundException(sagaId));

        SagaStep currentStep = saga.getCurrentStep()
            .orElseThrow(() -> new SagaExecutionException(sagaId,
                "NO_CURRENT_STEP", "No current step found"));

        if (!currentStep.getStepId().equals(stepId)) {
            throw new SagaExecutionException(sagaId, stepId,
                "STEP_MISMATCH", "Step ID does not match current step");
        }

        if (success) {
            saga.completeCurrentStep(output);
            saga = sagaRepository.save(saga);

            if (saga.getStatus() == SagaStatus.RUNNING) {
                executeNextStep(saga);
            } else if (saga.getStatus() == SagaStatus.COMPLETED) {
                eventPublisher.publishSagaCompleted(saga);
            }
        } else {
            handleStepFailure(saga, currentStep,
                new SagaExecutionException(sagaId, stepId, errorCode, errorMessage));
        }
    }
}
