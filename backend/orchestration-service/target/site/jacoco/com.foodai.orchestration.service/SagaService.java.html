package com.foodai.orchestration.service;

import com.foodai.orchestration.domain.model.Saga;
import com.foodai.orchestration.domain.model.SagaStatus;
import com.foodai.orchestration.domain.model.SagaStep;
import com.foodai.orchestration.domain.model.SagaType;
import com.foodai.orchestration.domain.repository.SagaRepository;
import com.foodai.orchestration.dto.request.SagaCallbackRequest;
import com.foodai.orchestration.dto.request.StartSagaRequest;
import com.foodai.orchestration.dto.response.SagaResponse;
import com.foodai.orchestration.exception.SagaExecutionException;
import com.foodai.orchestration.exception.SagaNotFoundException;
import com.foodai.orchestration.mapper.SagaMapper;
import com.foodai.orchestration.saga.definition.SagaDefinition;
import com.foodai.orchestration.saga.definition.SagaDefinitionRegistry;
import com.foodai.orchestration.saga.executor.SagaExecutor;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.Instant;
import java.time.temporal.ChronoUnit;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.UUID;

import static net.logstash.logback.argument.StructuredArguments.kv;

/**
 * Application service for saga management.
 *
 * @author FoodAI Team
 */
@Service
@RequiredArgsConstructor
@Slf4j
public class SagaService {

    private final SagaRepository sagaRepository;
    private final SagaMapper sagaMapper;
    private final SagaDefinitionRegistry definitionRegistry;
    private final SagaExecutor sagaExecutor;

    /**
     * Starts a new saga.
     *
     * @param request start saga request
     * @return saga response
     */
    @Transactional
    public SagaResponse startSaga(StartSagaRequest request) {
        log.info("Starting saga",
            kv("type", request.getType()),
            kv("userId", request.getUserId()),
            kv("entityId", request.getEntityId()));

        String correlationId = UUID.randomUUID().toString();

        // Get saga definition
        SagaDefinition definition = definitionRegistry.getDefinition(request.getType())
            .orElseThrow(() -> new IllegalArgumentException(
                "No saga definition found for type: " + request.getType()));

        // Create saga entity
        Saga saga = sagaMapper.toEntity(request, correlationId);
        saga.setSagaDefinition(definition.getName());
        saga.setContext(new HashMap<>(request.getPayload() != null ? request.getPayload() : Map.of()));

        // Add steps from definition
        definition.getSteps().forEach(stepDef -> {
            SagaStep step = SagaStep.builder()
                .stepId(UUID.randomUUID().toString())
                .stepName(stepDef.getName())
                .targetService(stepDef.getTargetService())
                .action(stepDef.getAction())
                .compensationAction(stepDef.getCompensationAction())
                .status(com.foodai.orchestration.domain.model.SagaStepStatus.PENDING)
                .compensatable(stepDef.isCompensatable())
                .maxRetries(stepDef.getMaxRetries())
                .timeoutMs(stepDef.getTimeoutMs())
                .build();
            saga.addStep(step);
        });

        saga.validate();
        Saga saved = sagaRepository.save(saga);

        log.info("Saga created successfully",
            kv("sagaId", saved.getId()),
            kv("correlationId", correlationId),
            kv("type", request.getType()));

        // Execute saga asynchronously
        executeSagaAsync(saved.getId());

        return sagaMapper.toResponse(saved);
    }

    /**
     * Executes a saga asynchronously.
     *
     * @param sagaId saga ID
     */
    @Async
    public void executeSagaAsync(String sagaId) {
        try {
            sagaExecutor.execute(sagaId);
        } catch (Exception e) {
            log.error("Async saga execution failed",
                kv("sagaId", sagaId), e);
        }
    }

    /**
     * Gets a saga by ID.
     *
     * @param sagaId saga ID
     * @return saga response
     */
    @Transactional(readOnly = true)
    public SagaResponse getSaga(String sagaId) {
        log.debug("Getting saga", kv("sagaId", sagaId));
        Saga saga = sagaRepository.findById(sagaId)
            .orElseThrow(() -> new SagaNotFoundException(sagaId));
        return sagaMapper.toResponse(saga);
    }

    /**
     * Gets a saga by correlation ID.
     *
     * @param correlationId correlation ID
     * @return saga response
     */
    @Transactional(readOnly = true)
    public SagaResponse getSagaByCorrelationId(String correlationId) {
        log.debug("Getting saga by correlation ID", kv("correlationId", correlationId));
        Saga saga = sagaRepository.findByCorrelationId(correlationId)
            .orElseThrow(() -> new SagaNotFoundException("correlation:" + correlationId));
        return sagaMapper.toResponse(saga);
    }

    /**
     * Gets sagas by user ID.
     *
     * @param userId user ID
     * @param pageable pagination
     * @return page of saga responses
     */
    @Transactional(readOnly = true)
    public Page<SagaResponse> getSagasByUserId(String userId, Pageable pageable) {
        log.debug("Getting sagas by user", kv("userId", userId));
        return sagaRepository.findByUserId(userId, pageable)
            .map(sagaMapper::toResponse);
    }

    /**
     * Gets sagas by type and status.
     *
     * @param type saga type
     * @param status saga status
     * @param pageable pagination
     * @return page of saga responses
     */
    @Transactional(readOnly = true)
    public Page<SagaResponse> getSagasByTypeAndStatus(
            SagaType type, SagaStatus status, Pageable pageable) {
        log.debug("Getting sagas by type and status",
            kv("type", type), kv("status", status));
        return sagaRepository.findByTypeAndStatus(type, status, pageable)
            .map(sagaMapper::toResponse);
    }

    /**
     * Handles a saga step callback.
     *
     * @param request callback request
     * @return updated saga response
     */
    @Transactional
    public SagaResponse handleCallback(SagaCallbackRequest request) {
        log.info("Handling saga callback",
            kv("sagaId", request.getSagaId()),
            kv("stepId", request.getStepId()),
            kv("success", request.isSuccess()));

        Saga saga = sagaRepository.findById(request.getSagaId())
            .orElseThrow(() -> new SagaNotFoundException(request.getSagaId()));

        if (saga.getStatus() != SagaStatus.WAITING) {
            throw new SagaExecutionException(request.getSagaId(),
                "INVALID_STATE",
                "Saga is not waiting for callback");
        }

        SagaStep currentStep = saga.getCurrentStep()
            .orElseThrow(() -> new SagaExecutionException(request.getSagaId(),
                "NO_CURRENT_STEP",
                "No current step found"));

        if (!currentStep.getStepId().equals(request.getStepId())) {
            throw new SagaExecutionException(request.getSagaId(), request.getStepId(),
                "STEP_MISMATCH",
                "Callback step ID does not match current step");
        }

        if (request.isSuccess()) {
            saga.completeCurrentStep(request.getOutputData());
            saga = sagaRepository.save(saga);

            // Continue execution if not complete
            if (saga.getStatus() == SagaStatus.RUNNING) {
                executeSagaAsync(saga.getId());
            }
        } else {
            saga.failCurrentStep(request.getErrorCode(), request.getErrorMessage());
            saga = sagaRepository.save(saga);
        }

        log.info("Saga callback processed",
            kv("sagaId", saga.getId()),
            kv("status", saga.getStatus()));

        return sagaMapper.toResponse(saga);
    }

    /**
     * Cancels a saga.
     *
     * @param sagaId saga ID
     * @return updated saga response
     */
    @Transactional
    public SagaResponse cancelSaga(String sagaId) {
        log.info("Cancelling saga", kv("sagaId", sagaId));

        Saga saga = sagaRepository.findById(sagaId)
            .orElseThrow(() -> new SagaNotFoundException(sagaId));

        if (saga.isTerminal()) {
            throw new IllegalStateException("Cannot cancel a terminal saga");
        }

        saga.cancel();
        saga = sagaRepository.save(saga);

        log.info("Saga cancelled", kv("sagaId", sagaId));
        return sagaMapper.toResponse(saga);
    }

    /**
     * Retries a failed saga.
     *
     * @param sagaId saga ID
     * @return updated saga response
     */
    @Transactional
    public SagaResponse retrySaga(String sagaId) {
        log.info("Retrying saga", kv("sagaId", sagaId));

        Saga saga = sagaRepository.findById(sagaId)
            .orElseThrow(() -> new SagaNotFoundException(sagaId));

        if (!saga.canRetry()) {
            throw new IllegalStateException("Saga cannot be retried");
        }

        saga.retry();
        saga = sagaRepository.save(saga);

        // Execute saga asynchronously
        executeSagaAsync(saga.getId());

        log.info("Saga retry initiated",
            kv("sagaId", sagaId),
            kv("retryCount", saga.getRetryCount()));

        return sagaMapper.toResponse(saga);
    }

    /**
     * Finds and recovers stuck sagas.
     */
    @Transactional
    public void recoverStuckSagas() {
        log.info("Checking for stuck sagas");

        Instant threshold = Instant.now().minus(5, ChronoUnit.MINUTES);
        List<Saga> stuckSagas = sagaRepository.findStuckSagas(SagaStatus.RUNNING, threshold);

        for (Saga saga : stuckSagas) {
            log.warn("Found stuck saga",
                kv("sagaId", saga.getId()),
                kv("startedAt", saga.getStartedAt()));

            if (saga.canRetry()) {
                saga.retry();
                sagaRepository.save(saga);
                executeSagaAsync(saga.getId());
            } else {
                saga.fail("STUCK_SAGA", "Saga stuck and max retries exceeded");
                sagaRepository.save(saga);
            }
        }

        log.info("Stuck saga recovery completed", kv("count", stuckSagas.size()));
    }

    /**
     * Gets saga statistics.
     *
     * @param type saga type (optional)
     * @return statistics map
     */
    @Transactional(readOnly = true)
    public Map<String, Object> getStatistics(SagaType type) {
        Map<String, Object> stats = new HashMap<>();

        if (type != null) {
            stats.put("pending", sagaRepository.countByTypeAndStatus(type, SagaStatus.PENDING));
            stats.put("running", sagaRepository.countByTypeAndStatus(type, SagaStatus.RUNNING));
            stats.put("completed", sagaRepository.countByTypeAndStatus(type, SagaStatus.COMPLETED));
            stats.put("failed", sagaRepository.countByTypeAndStatus(type, SagaStatus.FAILED));
            stats.put("compensated", sagaRepository.countByTypeAndStatus(type, SagaStatus.COMPENSATED));
        } else {
            for (SagaStatus status : SagaStatus.values()) {
                stats.put(status.name().toLowerCase(),
                    sagaRepository.findByStatus(status).size());
            }
        }

        return stats;
    }
}
