package com.foodai.order.controller;

import com.foodai.order.dto.request.InitiatePaymentRequest;
import com.foodai.order.dto.request.InitiateRefundRequest;
import com.foodai.order.dto.request.VerifyPaymentRequest;
import com.foodai.order.dto.response.*;
import com.foodai.order.service.PaymentService;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.Parameter;
import io.swagger.v3.oas.annotations.responses.ApiResponses;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.List;

/**
 * REST controller for payment and refund operations.
 *
 * @author FoodAI Team
 */
@RestController
@RequestMapping("/api/v1/payments")
@RequiredArgsConstructor
@Slf4j
@Tag(name = "Payment Management", description = "APIs for managing payments and refunds")
public class PaymentController {

    private final PaymentService paymentService;

    /**
     * Initiates a payment.
     */
    @PostMapping("/initiate")
    @Operation(summary = "Initiate Payment", description = "Initiates a payment for an order")
    @ApiResponses(value = {
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "201", description = "Payment initiated successfully"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "400", description = "Invalid request")
    })
    public ResponseEntity<ApiResponse<PaymentInitiationResponse>> initiatePayment(
            @Valid @RequestBody InitiatePaymentRequest request) {
        log.info("Request to initiate payment for order: {}", request.getOrderId());
        PaymentInitiationResponse response = paymentService.initiatePayment(request);
        return ResponseEntity
            .status(HttpStatus.CREATED)
            .body(ApiResponse.success("Payment initiated", response));
    }

    /**
     * Verifies a payment.
     */
    @PostMapping("/verify")
    @Operation(summary = "Verify Payment", description = "Verifies a payment from the payment gateway")
    @ApiResponses(value = {
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "200", description = "Payment verified successfully"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "400", description = "Payment verification failed"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "402", description = "Payment failed")
    })
    public ResponseEntity<ApiResponse<PaymentResponse>> verifyPayment(
            @Valid @RequestBody VerifyPaymentRequest request) {
        log.info("Request to verify payment for gateway order: {}", request.getGatewayOrderId());
        PaymentResponse response = paymentService.verifyPayment(request);
        return ResponseEntity.ok(ApiResponse.success("Payment verified successfully", response));
    }

    /**
     * Gets a payment by ID.
     */
    @GetMapping("/{paymentId}")
    @Operation(summary = "Get Payment", description = "Gets a payment by its ID")
    @ApiResponses(value = {
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "200", description = "Payment found"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "404", description = "Payment not found")
    })
    public ResponseEntity<ApiResponse<PaymentResponse>> getPayment(
            @Parameter(description = "Payment ID") @PathVariable String paymentId) {
        log.info("Request to get payment: {}", paymentId);
        PaymentResponse response = paymentService.getPayment(paymentId);
        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Gets payment by order ID.
     */
    @GetMapping("/order/{orderId}")
    @Operation(summary = "Get Payment by Order", description = "Gets the payment for an order")
    @ApiResponses(value = {
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "200", description = "Payment found"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "404", description = "Payment not found")
    })
    public ResponseEntity<ApiResponse<PaymentResponse>> getPaymentByOrderId(
            @Parameter(description = "Order ID") @PathVariable String orderId) {
        log.info("Request to get payment for order: {}", orderId);
        PaymentResponse response = paymentService.getPaymentByOrderId(orderId);
        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Gets payments for a user.
     */
    @GetMapping("/user/{userId}")
    @Operation(summary = "Get User Payments", description = "Gets all payments for a user")
    public ResponseEntity<ApiResponse<Page<PaymentResponse>>> getUserPayments(
            @Parameter(description = "User ID") @PathVariable String userId,
            @Parameter(description = "Page number") @RequestParam(defaultValue = "0") int page,
            @Parameter(description = "Page size") @RequestParam(defaultValue = "20") int size) {
        log.info("Request to get payments for user: {}", userId);
        Pageable pageable = PageRequest.of(page, size, Sort.by("createdAt").descending());
        Page<PaymentResponse> response = paymentService.getUserPayments(userId, pageable);
        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Initiates a refund.
     */
    @PostMapping("/refunds")
    @Operation(summary = "Initiate Refund", description = "Initiates a refund for an order")
    @ApiResponses(value = {
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "201", description = "Refund initiated successfully"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "400", description = "Invalid request"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "409", description = "Cannot refund")
    })
    public ResponseEntity<ApiResponse<RefundResponse>> initiateRefund(
            @Valid @RequestBody InitiateRefundRequest request) {
        log.info("Request to initiate refund for order: {}", request.getOrderId());
        RefundResponse response = paymentService.initiateRefund(request);
        return ResponseEntity
            .status(HttpStatus.CREATED)
            .body(ApiResponse.success("Refund initiated", response));
    }

    /**
     * Gets a refund by ID.
     */
    @GetMapping("/refunds/{refundId}")
    @Operation(summary = "Get Refund", description = "Gets a refund by its ID")
    @ApiResponses(value = {
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "200", description = "Refund found"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "404", description = "Refund not found")
    })
    public ResponseEntity<ApiResponse<RefundResponse>> getRefund(
            @Parameter(description = "Refund ID") @PathVariable String refundId) {
        log.info("Request to get refund: {}", refundId);
        RefundResponse response = paymentService.getRefund(refundId);
        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Gets refunds for an order.
     */
    @GetMapping("/refunds/order/{orderId}")
    @Operation(summary = "Get Order Refunds", description = "Gets all refunds for an order")
    public ResponseEntity<ApiResponse<List<RefundResponse>>> getOrderRefunds(
            @Parameter(description = "Order ID") @PathVariable String orderId) {
        log.info("Request to get refunds for order: {}", orderId);
        List<RefundResponse> response = paymentService.getRefundsByOrderId(orderId);
        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Gets refunds for a user.
     */
    @GetMapping("/refunds/user/{userId}")
    @Operation(summary = "Get User Refunds", description = "Gets all refunds for a user")
    public ResponseEntity<ApiResponse<Page<RefundResponse>>> getUserRefunds(
            @Parameter(description = "User ID") @PathVariable String userId,
            @Parameter(description = "Page number") @RequestParam(defaultValue = "0") int page,
            @Parameter(description = "Page size") @RequestParam(defaultValue = "20") int size) {
        log.info("Request to get refunds for user: {}", userId);
        Pageable pageable = PageRequest.of(page, size, Sort.by("createdAt").descending());
        Page<RefundResponse> response = paymentService.getUserRefunds(userId, pageable);
        return ResponseEntity.ok(ApiResponse.success(response));
    }
}
