        tATConfigResponse.considerDistance( config.getConsiderDistance() );
        tATConfigResponse.maxTATMinutes( config.getMaxTATMinutes() );

    }

    @Override
    public void updateEntityFromRequest(Restaurant restaurant, CreateRestaurantRequest request) {
        if ( request == null ) {
            return;
        }

        if ( request.getName() != null ) {
            restaurant.setName( request.getName() );
        }
        if ( request.getDescription() != null ) {
            restaurant.setDescription( request.getDescription() );
        }
        if ( request.getLogo() != null ) {
            restaurant.setLogo( request.getLogo() );
        }
        if ( request.getCoverImage() != null ) {
            restaurant.setCoverImage( request.getCoverImage() );
        }
        if ( restaurant.getOwners() != null ) {
            List<OwnerVO> list = toOwnerVOList( request.getOwners() );
            if ( list != null ) {
                restaurant.getOwners().clear();
                restaurant.getOwners().addAll( list );
            }
        }
        else {
            List<OwnerVO> list = toOwnerVOList( request.getOwners() );
            if ( list != null ) {
                restaurant.setOwners( list );
            }
        }
        if ( restaurant.getContacts() != null ) {
            List<ContactVO> list1 = toContactVOList( request.getContacts() );
            if ( list1 != null ) {
                restaurant.getContacts().clear();
                restaurant.getContacts().addAll( list1 );
            }
        }
        else {
            List<ContactVO> list1 = toContactVOList( request.getContacts() );
            if ( list1 != null ) {
                restaurant.setContacts( list1 );
            }
        }
        if ( restaurant.getCuisineTypes() != null ) {
            List<String> list2 = request.getCuisineTypes();
            if ( list2 != null ) {
                restaurant.getCuisineTypes().clear();
                restaurant.getCuisineTypes().addAll( list2 );
            }
        }
        else {
            List<String> list2 = request.getCuisineTypes();
            if ( list2 != null ) {
                restaurant.setCuisineTypes( new ArrayList<String>( list2 ) );
            }
        }
        if ( restaurant.getDocuments() != null ) {
            List<DocumentVO> list3 = toDocumentVOList( request.getDocuments() );
            if ( list3 != null ) {
                restaurant.getDocuments().clear();
                restaurant.getDocuments().addAll( list3 );
            }
        }
        else {
            List<DocumentVO> list3 = toDocumentVOList( request.getDocuments() );
            if ( list3 != null ) {
                restaurant.setDocuments( list3 );
            }
        }
        if ( request.getContract() != null ) {
            restaurant.setContract( toContractVO( request.getContract() ) );
        }

        restaurant.setUpdatedAt( now() );
    }
