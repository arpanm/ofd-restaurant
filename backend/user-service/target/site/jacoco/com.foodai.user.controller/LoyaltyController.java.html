package com.foodai.user.controller;

import com.foodai.user.domain.model.LoyaltyTier;
import com.foodai.user.dto.response.ApiResponse;
import com.foodai.user.dto.response.LoyaltyAccountResponse;
import com.foodai.user.service.LoyaltyService;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.Parameter;
import io.swagger.v3.oas.annotations.responses.ApiResponses;
import io.swagger.v3.oas.annotations.tags.Tag;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.data.web.PageableDefault;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

/**
 * REST controller for Loyalty operations.
 * Provides endpoints for managing loyalty points, tiers, and rewards.
 *
 * @author FoodAI Team
 */
@RestController
@RequestMapping("/api/v1/loyalty")
@RequiredArgsConstructor
@Slf4j
@Tag(name = "Loyalty Management", description = "APIs for managing loyalty points, tiers, achievements, and rewards")
public class LoyaltyController {

    private final LoyaltyService loyaltyService;

    /**
     * Creates a loyalty account for a user.
     *
     * @param userId the user ID
     * @return the created loyalty account
     */
    @PostMapping("/accounts")
    @Operation(summary = "Create loyalty account", description = "Create a new loyalty account for a user")
    @ApiResponses(value = {
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "201", description = "Account created successfully"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "409", description = "Account already exists")
    })
    public ResponseEntity<ApiResponse<LoyaltyAccountResponse>> createAccount(
            @Parameter(description = "User ID") @RequestParam String userId) {
        log.info("POST /api/v1/loyalty/accounts - Creating account for user: {}", userId);

        LoyaltyAccountResponse response = loyaltyService.createAccount(userId);
        return ResponseEntity
            .status(HttpStatus.CREATED)
            .body(ApiResponse.success("Loyalty account created successfully", response));
    }

    /**
     * Gets a loyalty account by user ID.
     *
     * @param userId the user ID
     * @return the loyalty account
     */
    @GetMapping("/accounts/user/{userId}")
    @Operation(summary = "Get loyalty account by user", description = "Retrieve loyalty account for a user")
    @ApiResponses(value = {
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "200", description = "Account found"),
        @io.swagger.v3.oas.annotations.responses.ApiResponse(responseCode = "404", description = "Account not found")
    })
    public ResponseEntity<ApiResponse<LoyaltyAccountResponse>> getAccountByUserId(
            @Parameter(description = "User ID") @PathVariable String userId) {
        log.info("GET /api/v1/loyalty/accounts/user/{}", userId);

        LoyaltyAccountResponse response = loyaltyService.getAccountByUserId(userId);
        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Gets a loyalty account by ID.
     *
     * @param id the account ID
     * @return the loyalty account
     */
    @GetMapping("/accounts/{id}")
    @Operation(summary = "Get loyalty account by ID", description = "Retrieve loyalty account by ID")
    public ResponseEntity<ApiResponse<LoyaltyAccountResponse>> getAccountById(
            @Parameter(description = "Account ID") @PathVariable String id) {
        log.info("GET /api/v1/loyalty/accounts/{}", id);

        LoyaltyAccountResponse response = loyaltyService.getAccountById(id);
        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Earns points for an order.
     *
     * @param userId      the user ID
     * @param orderAmount the order amount
     * @param orderId     the order ID
     * @return the updated loyalty account
     */
    @PostMapping("/accounts/user/{userId}/earn")
    @Operation(summary = "Earn points", description = "Earn loyalty points for an order")
    public ResponseEntity<ApiResponse<LoyaltyAccountResponse>> earnPoints(
            @Parameter(description = "User ID") @PathVariable String userId,
            @Parameter(description = "Order amount in INR") @RequestParam double orderAmount,
            @Parameter(description = "Order ID") @RequestParam String orderId) {
        log.info("POST /api/v1/loyalty/accounts/user/{}/earn - amount: {}", userId, orderAmount);

        LoyaltyAccountResponse response = loyaltyService.earnPointsForOrder(userId, orderAmount, orderId);
        return ResponseEntity.ok(ApiResponse.success("Points earned successfully", response));
    }

    /**
     * Redeems points.
     *
     * @param userId the user ID
     * @param points the points to redeem
     * @param reason the redemption reason
     * @return the updated loyalty account
     */
    @PostMapping("/accounts/user/{userId}/redeem")
    @Operation(summary = "Redeem points", description = "Redeem loyalty points")
    public ResponseEntity<ApiResponse<LoyaltyAccountResponse>> redeemPoints(
            @Parameter(description = "User ID") @PathVariable String userId,
            @Parameter(description = "Points to redeem") @RequestParam long points,
            @Parameter(description = "Redemption reason") @RequestParam String reason) {
        log.info("POST /api/v1/loyalty/accounts/user/{}/redeem - points: {}", userId, points);

        LoyaltyAccountResponse response = loyaltyService.redeemPoints(userId, points, reason);
        return ResponseEntity.ok(ApiResponse.success("Points redeemed successfully", response));
    }

    /**
     * Processes a referral.
     *
     * @param referralCode the referral code
     * @param newUserId    the new user ID
     * @return the referrer's updated loyalty account
     */
    @PostMapping("/referrals/process")
    @Operation(summary = "Process referral", description = "Process a referral and award bonus points")
    public ResponseEntity<ApiResponse<LoyaltyAccountResponse>> processReferral(
            @Parameter(description = "Referral code") @RequestParam String referralCode,
            @Parameter(description = "New user ID") @RequestParam String newUserId) {
        log.info("POST /api/v1/loyalty/referrals/process - code: {}, newUser: {}", referralCode, newUserId);

        LoyaltyAccountResponse response = loyaltyService.processReferral(referralCode, newUserId);
        return ResponseEntity.ok(ApiResponse.success("Referral processed successfully", response));
    }

    /**
     * Gets accounts by tier.
     *
     * @param tier     the loyalty tier
     * @param pageable pagination information
     * @return page of loyalty accounts
     */
    @GetMapping("/accounts/tier/{tier}")
    @Operation(summary = "Get accounts by tier", description = "Get all accounts in a specific tier")
    public ResponseEntity<ApiResponse<Page<LoyaltyAccountResponse>>> getAccountsByTier(
            @Parameter(description = "Loyalty tier") @PathVariable LoyaltyTier tier,
            @PageableDefault(size = 20) Pageable pageable) {
        log.info("GET /api/v1/loyalty/accounts/tier/{}", tier);

        Page<LoyaltyAccountResponse> response = loyaltyService.getAccountsByTier(tier, pageable);
        return ResponseEntity.ok(ApiResponse.success(response));
    }

    /**
     * Gets the leaderboard.
     *
     * @param pageable pagination information
     * @return page of top loyalty accounts
     */
    @GetMapping("/leaderboard")
    @Operation(summary = "Get leaderboard", description = "Get top loyalty accounts by points")
    public ResponseEntity<ApiResponse<Page<LoyaltyAccountResponse>>> getLeaderboard(
            @PageableDefault(size = 20, sort = "currentPoints", direction = Sort.Direction.DESC) Pageable pageable) {
        log.info("GET /api/v1/loyalty/leaderboard");

        Page<LoyaltyAccountResponse> response = loyaltyService.getLeaderboard(pageable);
        return ResponseEntity.ok(ApiResponse.success(response));
    }
}
