package com.foodai.user.domain.model;

import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;
import org.springframework.data.annotation.CreatedDate;
import org.springframework.data.annotation.Id;
import org.springframework.data.annotation.LastModifiedDate;
import org.springframework.data.mongodb.core.index.Indexed;
import org.springframework.data.mongodb.core.mapping.Document;

import java.time.Instant;
import java.util.ArrayList;
import java.util.List;

/**
 * Loyalty Account Entity.
 * Manages user's loyalty points, tier status, achievements, and rewards.
 *
 * <p>Covers features from:
 * <ul>
 *   <li>LOYALTY_GAMIFICATION.md - Points, tiers, achievements, rewards</li>
 * </ul>
 *
 * @author FoodAI Team
 */
@Document(collection = "loyalty_accounts")
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class LoyaltyAccount {

    /**
     * Unique loyalty account identifier
     */
    @Id
    private String id;

    /**
     * User ID this account belongs to
     */
    @Indexed(unique = true)
    private String userId;

    /**
     * Current loyalty points balance
     */
    @Builder.Default
    private Long currentPoints = 0L;

    /**
     * Total points earned (lifetime)
     */
    @Builder.Default
    private Long lifetimePointsEarned = 0L;

    /**
     * Total points redeemed (lifetime)
     */
    @Builder.Default
    private Long lifetimePointsRedeemed = 0L;

    /**
     * Total points expired (lifetime)
     */
    @Builder.Default
    private Long lifetimePointsExpired = 0L;

    /**
     * Current loyalty tier
     */
    @Indexed
    @Builder.Default
    private LoyaltyTier tier = LoyaltyTier.BRONZE;

    /**
     * Points needed to reach next tier
     */
    @Builder.Default
    private Long pointsToNextTier = 1000L;

    /**
     * Points earned in current tier period (for tier maintenance)
     */
    @Builder.Default
    private Long tierPeriodPoints = 0L;

    /**
     * Tier period start date
     */
    private Instant tierPeriodStart;

    /**
     * Tier period end date
     */
    private Instant tierPeriodEnd;

    /**
     * Next tier downgrade review date
     */
    private Instant nextTierReviewDate;

    /**
     * Current streak days (consecutive days with orders)
     */
    @Builder.Default
    private Integer currentStreak = 0;

    /**
     * Longest streak ever
     */
    @Builder.Default
    private Integer longestStreak = 0;

    /**
     * Last order date (for streak calculation)
     */
    private Instant lastOrderDate;

    /**
     * Referral code for this user
     */
    @Indexed(unique = true, sparse = true)
    private String referralCode;

    /**
     * Total successful referrals
     */
    @Builder.Default
    private Integer totalReferrals = 0;

    /**
     * Points earned from referrals
     */
    @Builder.Default
    private Long referralPointsEarned = 0L;

    /**
     * Unlocked achievements
     */
    @Builder.Default
    private List<AchievementVO> achievements = new ArrayList<>();

    /**
     * Unlocked badges
     */
    @Builder.Default
    private List<BadgeVO> badges = new ArrayList<>();

    /**
     * Active challenges
     */
    @Builder.Default
    private List<ChallengeProgressVO> activeChallenges = new ArrayList<>();

    /**
     * Account status
     */
    @Builder.Default
    private LoyaltyAccountStatus status = LoyaltyAccountStatus.ACTIVE;

    /**
     * Creation timestamp
     */
    @CreatedDate
    private Instant createdAt;

    /**
     * Last update timestamp
     */
    @LastModifiedDate
    private Instant updatedAt;

    // ==================== Domain Methods ====================

    /**
     * Earns points for the account.
     *
     * @param points the points to earn
     * @param source the source of points
     */
    public void earnPoints(long points, String source) {
        this.currentPoints += points;
        this.lifetimePointsEarned += points;
        this.tierPeriodPoints += points;
        checkTierUpgrade();
    }

    /**
     * Redeems points from the account.
     *
     * @param points the points to redeem
     * @return true if redemption was successful
     */
    public boolean redeemPoints(long points) {
        if (this.currentPoints < points) {
            return false;
        }
        this.currentPoints -= points;
        this.lifetimePointsRedeemed += points;
        return true;
    }

    /**
     * Expires points from the account.
     *
     * @param points the points to expire
     */
    public void expirePoints(long points) {
        this.currentPoints = Math.max(0, this.currentPoints - points);
        this.lifetimePointsExpired += points;
    }

    /**
     * Updates streak based on new order.
     *
     * @param orderDate the order date
     */
    public void updateStreak(Instant orderDate) {
        if (lastOrderDate == null) {
            this.currentStreak = 1;
        } else {
            // Check if order is on consecutive day
            long daysBetween = java.time.Duration.between(lastOrderDate, orderDate).toDays();
            if (daysBetween == 1) {
                this.currentStreak++;
            } else if (daysBetween > 1) {
                this.currentStreak = 1;
            }
            // Same day doesn't change streak
        }
        this.lastOrderDate = orderDate;
        if (this.currentStreak > this.longestStreak) {
            this.longestStreak = this.currentStreak;
        }
    }

    /**
     * Checks and applies tier upgrade if eligible.
     */
    private void checkTierUpgrade() {
        LoyaltyTier newTier = LoyaltyTier.getTierByPoints(this.tierPeriodPoints);
        if (newTier.ordinal() > this.tier.ordinal()) {
            this.tier = newTier;
        }
        this.pointsToNextTier = calculatePointsToNextTier();
    }

    /**
     * Calculates points needed to reach next tier.
     *
     * @return points to next tier
     */
    private long calculatePointsToNextTier() {
        return switch (this.tier) {
            case BRONZE -> 1000 - this.tierPeriodPoints;
            case SILVER -> 5000 - this.tierPeriodPoints;
            case GOLD -> 15000 - this.tierPeriodPoints;
            case PLATINUM -> 50000 - this.tierPeriodPoints;
            case DIAMOND -> 0; // Max tier
        };
    }

    /**
     * Adds an achievement to the account.
     *
     * @param achievement the achievement to add
     */
    public void addAchievement(AchievementVO achievement) {
        if (this.achievements == null) {
            this.achievements = new ArrayList<>();
        }
        // Check if already unlocked
        boolean alreadyUnlocked = this.achievements.stream()
            .anyMatch(a -> a.getAchievementId().equals(achievement.getAchievementId()));
        if (!alreadyUnlocked) {
            this.achievements.add(achievement);
            if (achievement.getPointsAwarded() != null) {
                earnPoints(achievement.getPointsAwarded(), "achievement:" + achievement.getAchievementId());
            }
        }
    }

    /**
     * Adds a badge to the account.
     *
     * @param badge the badge to add
     */
    public void addBadge(BadgeVO badge) {
        if (this.badges == null) {
            this.badges = new ArrayList<>();
        }
        boolean alreadyHas = this.badges.stream()
            .anyMatch(b -> b.getBadgeId().equals(badge.getBadgeId()));
        if (!alreadyHas) {
            this.badges.add(badge);
        }
    }

    /**
     * Records a successful referral.
     *
     * @param referralPoints points earned from referral
     */
    public void recordReferral(long referralPoints) {
        this.totalReferrals++;
        this.referralPointsEarned += referralPoints;
        earnPoints(referralPoints, "referral");
    }

    /**
     * Gets tier benefits multiplier.
     *
     * @return points multiplier based on tier
     */
    public double getTierMultiplier() {
        return this.tier.getPointsMultiplier();
    }
}
