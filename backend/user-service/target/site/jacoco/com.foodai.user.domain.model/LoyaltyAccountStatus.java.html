package com.foodai.user.domain.model;

/**
 * Enumeration of loyalty account statuses.
 *
 * @author FoodAI Team
 */
public enum LoyaltyAccountStatus {
    /**
     * Account is active and fully functional
     */
    ACTIVE,

    /**
     * Account is suspended due to fraud or violations
     */
    SUSPENDED,

    /**
     * Account is frozen pending investigation
     */
    FROZEN,

    /**
     * Account is closed
     */
    CLOSED
}
