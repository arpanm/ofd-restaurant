package com.foodai.user.domain.model;

import lombok.Getter;

/**
 * Enumeration of loyalty program tiers with associated benefits.
 *
 * @author FoodAI Team
 */
@Getter
public enum LoyaltyTier {
    /**
     * Bronze tier - Entry level (0-999 points)
     */
    BRONZE(0, 999, 1.0, 0, 0),

    /**
     * Silver tier - 1000-4999 points
     */
    SILVER(1000, 4999, 1.25, 5, 1),

    /**
     * Gold tier - 5000-14999 points
     */
    GOLD(5000, 14999, 1.5, 10, 2),

    /**
     * Platinum tier - 15000-49999 points
     */
    PLATINUM(15000, 49999, 1.75, 15, 3),

    /**
     * Diamond tier - 50000+ points
     */
    DIAMOND(50000, Long.MAX_VALUE, 2.0, 20, 5);

    /**
     * Minimum points required for this tier
     */
    private final long minPoints;

    /**
     * Maximum points for this tier
     */
    private final long maxPoints;

    /**
     * Points earning multiplier
     */
    private final double pointsMultiplier;

    /**
     * Discount percentage on orders
     */
    private final int discountPercentage;

    /**
     * Free deliveries per month
     */
    private final int freeDeliveriesPerMonth;

    LoyaltyTier(long minPoints, long maxPoints, double pointsMultiplier,
                int discountPercentage, int freeDeliveriesPerMonth) {
        this.minPoints = minPoints;
        this.maxPoints = maxPoints;
        this.pointsMultiplier = pointsMultiplier;
        this.discountPercentage = discountPercentage;
        this.freeDeliveriesPerMonth = freeDeliveriesPerMonth;
    }

    /**
     * Gets the tier based on points.
     *
     * @param points the points count
     * @return the corresponding tier
     */
    public static LoyaltyTier getTierByPoints(long points) {
        if (points >= DIAMOND.minPoints) return DIAMOND;
        if (points >= PLATINUM.minPoints) return PLATINUM;
        if (points >= GOLD.minPoints) return GOLD;
        if (points >= SILVER.minPoints) return SILVER;
        return BRONZE;
    }
}
