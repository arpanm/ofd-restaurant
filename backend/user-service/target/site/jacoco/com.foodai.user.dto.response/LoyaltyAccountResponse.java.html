package com.foodai.user.dto.response;

import com.foodai.user.domain.model.LoyaltyAccountStatus;
import com.foodai.user.domain.model.LoyaltyTier;
import io.swagger.v3.oas.annotations.media.Schema;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

import java.time.Instant;
import java.util.List;

/**
 * Response DTO for Loyalty Account.
 *
 * @author FoodAI Team
 */
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
@Schema(description = "Loyalty account response")
public class LoyaltyAccountResponse {

    @Schema(description = "Account ID")
    private String id;

    @Schema(description = "User ID")
    private String userId;

    @Schema(description = "Current points balance")
    private Long currentPoints;

    @Schema(description = "Lifetime points earned")
    private Long lifetimePointsEarned;

    @Schema(description = "Lifetime points redeemed")
    private Long lifetimePointsRedeemed;

    @Schema(description = "Current tier")
    private LoyaltyTier tier;

    @Schema(description = "Tier name")
    private String tierName;

    @Schema(description = "Points to next tier")
    private Long pointsToNextTier;

    @Schema(description = "Tier progress percentage")
    private Integer tierProgressPercentage;

    @Schema(description = "Points multiplier")
    private Double pointsMultiplier;

    @Schema(description = "Tier discount percentage")
    private Integer tierDiscountPercentage;

    @Schema(description = "Free deliveries per month")
    private Integer freeDeliveriesPerMonth;

    @Schema(description = "Current streak days")
    private Integer currentStreak;

    @Schema(description = "Longest streak ever")
    private Integer longestStreak;

    @Schema(description = "Referral code")
    private String referralCode;

    @Schema(description = "Total referrals")
    private Integer totalReferrals;

    @Schema(description = "Points from referrals")
    private Long referralPointsEarned;

    @Schema(description = "Unlocked achievements")
    private List<AchievementResponse> achievements;

    @Schema(description = "Unlocked badges")
    private List<BadgeResponse> badges;

    @Schema(description = "Active challenges")
    private List<ChallengeProgressResponse> activeChallenges;

    @Schema(description = "Account status")
    private LoyaltyAccountStatus status;

    @Schema(description = "Tier period start")
    private Instant tierPeriodStart;

    @Schema(description = "Tier period end")
    private Instant tierPeriodEnd;

    @Schema(description = "Created at")
    private Instant createdAt;
}
