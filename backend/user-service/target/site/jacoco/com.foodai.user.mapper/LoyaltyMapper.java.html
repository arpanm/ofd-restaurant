package com.foodai.user.mapper;

import com.foodai.user.domain.model.*;
import com.foodai.user.dto.response.*;
import org.mapstruct.*;

import java.util.List;

/**
 * MapStruct mapper for Loyalty entities and DTOs.
 *
 * @author FoodAI Team
 */
@Mapper(componentModel = MappingConstants.ComponentModel.SPRING)
public interface LoyaltyMapper {

    /**
     * Maps LoyaltyAccount entity to response DTO.
     *
     * @param account the loyalty account
     * @return the response DTO
     */
    @Mapping(target = "tierName", expression = "java(account.getTier().name())")
    @Mapping(target = "tierProgressPercentage", expression = "java(calculateTierProgress(account))")
    @Mapping(target = "pointsMultiplier", expression = "java(account.getTierMultiplier())")
    @Mapping(target = "tierDiscountPercentage", expression = "java(account.getTier().getDiscountPercentage())")
    @Mapping(target = "freeDeliveriesPerMonth", expression = "java(account.getTier().getFreeDeliveriesPerMonth())")
    LoyaltyAccountResponse toResponse(LoyaltyAccount account);

    /**
     * Maps list of LoyaltyAccount entities to response DTOs.
     *
     * @param accounts the list of accounts
     * @return list of response DTOs
     */
    List<LoyaltyAccountResponse> toResponseList(List<LoyaltyAccount> accounts);

    // ==================== Achievement Mappings ====================

    AchievementResponse toAchievementResponse(AchievementVO achievement);

    List<AchievementResponse> toAchievementResponseList(List<AchievementVO> achievements);

    // ==================== Badge Mappings ====================

    BadgeResponse toBadgeResponse(BadgeVO badge);

    List<BadgeResponse> toBadgeResponseList(List<BadgeVO> badges);

    // ==================== Challenge Mappings ====================

    @Mapping(target = "progressPercentage", expression = "java(challenge.getProgressPercentage())")
    @Mapping(target = "expired", expression = "java(challenge.isExpired())")
    ChallengeProgressResponse toChallengeProgressResponse(ChallengeProgressVO challenge);

    List<ChallengeProgressResponse> toChallengeProgressResponseList(List<ChallengeProgressVO> challenges);

    // ==================== Helper Methods ====================

    /**
     * Calculates tier progress percentage.
     *
     * @param account the loyalty account
     * @return progress percentage
     */
    default int calculateTierProgress(LoyaltyAccount account) {
        if (account.getTier() == LoyaltyTier.DIAMOND) {
            return 100;
        }
        long tierPoints = account.getTierPeriodPoints();
        long minPoints = account.getTier().getMinPoints();
        long maxPoints = account.getTier().getMaxPoints();
        
        if (maxPoints == Long.MAX_VALUE) {
            return 100;
        }
        
        long tierRange = maxPoints - minPoints;
        long pointsInTier = tierPoints - minPoints;
        
        return (int) Math.min(100, (pointsInTier * 100) / tierRange);
    }
}
