package com.foodai.user.service;

import com.foodai.user.domain.model.*;
import com.foodai.user.domain.repository.LoyaltyAccountRepository;
import com.foodai.user.dto.response.LoyaltyAccountResponse;
import com.foodai.user.exception.LoyaltyAccountNotFoundException;
import com.foodai.user.mapper.LoyaltyMapper;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.Instant;
import java.time.temporal.ChronoUnit;
import java.util.HashMap;
import java.util.Map;
import java.util.UUID;

/**
 * Service for Loyalty operations.
 * Handles loyalty points, tiers, achievements, and rewards.
 *
 * @author FoodAI Team
 */
@Service
@RequiredArgsConstructor
@Slf4j
@Transactional(readOnly = true)
public class LoyaltyService {

    private final LoyaltyAccountRepository loyaltyAccountRepository;
    private final LoyaltyMapper loyaltyMapper;
    private final KafkaTemplate<String, Object> kafkaTemplate;

    private static final long POINTS_PER_RUPEE = 1L;
    private static final long REFERRAL_BONUS_POINTS = 500L;
    private static final long REFEREE_BONUS_POINTS = 250L;

    /**
     * Creates a loyalty account for a user.
     *
     * @param userId the user ID
     * @return the created loyalty account response
     */
    @Transactional
    public LoyaltyAccountResponse createAccount(String userId) {
        log.info("Creating loyalty account for user: {}", userId);

        // Check if account already exists
        if (loyaltyAccountRepository.findByUserId(userId).isPresent()) {
            throw new IllegalStateException("Loyalty account already exists for user: " + userId);
        }

        // Create new account
        LoyaltyAccount account = LoyaltyAccount.builder()
            .userId(userId)
            .referralCode(generateReferralCode())
            .tier(LoyaltyTier.BRONZE)
            .tierPeriodStart(Instant.now())
            .tierPeriodEnd(Instant.now().plus(365, ChronoUnit.DAYS))
            .build();

        LoyaltyAccount saved = loyaltyAccountRepository.save(account);
        log.info("Loyalty account created with ID: {}", saved.getId());

        return loyaltyMapper.toResponse(saved);
    }

    /**
     * Gets a loyalty account by user ID.
     *
     * @param userId the user ID
     * @return the loyalty account response
     */
    public LoyaltyAccountResponse getAccountByUserId(String userId) {
        log.debug("Fetching loyalty account for user: {}", userId);

        LoyaltyAccount account = loyaltyAccountRepository.findByUserId(userId)
            .orElseThrow(() -> new LoyaltyAccountNotFoundException(userId));

        return loyaltyMapper.toResponse(account);
    }

    /**
     * Gets a loyalty account by ID.
     *
     * @param id the account ID
     * @return the loyalty account response
     */
    public LoyaltyAccountResponse getAccountById(String id) {
        log.debug("Fetching loyalty account with ID: {}", id);

        LoyaltyAccount account = loyaltyAccountRepository.findById(id)
            .orElseThrow(() -> new LoyaltyAccountNotFoundException("id", id));

        return loyaltyMapper.toResponse(account);
    }

    /**
     * Earns points for an order.
     *
     * @param userId      the user ID
     * @param orderAmount the order amount in INR
     * @param orderId     the order ID
     * @return the updated loyalty account response
     */
    @Transactional
    public LoyaltyAccountResponse earnPointsForOrder(String userId, double orderAmount, String orderId) {
        log.info("Earning points for user: {}, order: {}, amount: {}", userId, orderId, orderAmount);

        LoyaltyAccount account = loyaltyAccountRepository.findByUserId(userId)
            .orElseThrow(() -> new LoyaltyAccountNotFoundException(userId));

        // Calculate points with tier multiplier
        long basePoints = (long) (orderAmount * POINTS_PER_RUPEE);
        long earnedPoints = (long) (basePoints * account.getTierMultiplier());

        account.earnPoints(earnedPoints, "order:" + orderId);
        account.updateStreak(Instant.now());

        LoyaltyAccount updated = loyaltyAccountRepository.save(account);
        log.info("Earned {} points for user: {}", earnedPoints, userId);

        // Publish event
        publishLoyaltyEvent("loyalty.points.earned", updated, earnedPoints, "order");

        return loyaltyMapper.toResponse(updated);
    }

    /**
     * Redeems points.
     *
     * @param userId the user ID
     * @param points the points to redeem
     * @param reason the redemption reason
     * @return the updated loyalty account response
     */
    @Transactional
    public LoyaltyAccountResponse redeemPoints(String userId, long points, String reason) {
        log.info("Redeeming {} points for user: {}", points, userId);

        LoyaltyAccount account = loyaltyAccountRepository.findByUserId(userId)
            .orElseThrow(() -> new LoyaltyAccountNotFoundException(userId));

        if (!account.redeemPoints(points)) {
            throw new IllegalStateException("Insufficient points. Available: " + account.getCurrentPoints());
        }

        LoyaltyAccount updated = loyaltyAccountRepository.save(account);
        log.info("Redeemed {} points for user: {}", points, userId);

        // Publish event
        publishLoyaltyEvent("loyalty.points.redeemed", updated, points, reason);

        return loyaltyMapper.toResponse(updated);
    }

    /**
     * Processes a referral.
     *
     * @param referralCode the referral code
     * @param newUserId    the new user ID
     * @return the referrer's updated loyalty account response
     */
    @Transactional
    public LoyaltyAccountResponse processReferral(String referralCode, String newUserId) {
        log.info("Processing referral with code: {} for new user: {}", referralCode, newUserId);

        // Find referrer's account
        LoyaltyAccount referrerAccount = loyaltyAccountRepository.findByReferralCode(referralCode)
            .orElseThrow(() -> new IllegalArgumentException("Invalid referral code: " + referralCode));

        // Award points to referrer
        referrerAccount.recordReferral(REFERRAL_BONUS_POINTS);
        loyaltyAccountRepository.save(referrerAccount);

        // Award points to referee (new user)
        loyaltyAccountRepository.findByUserId(newUserId).ifPresent(refereeAccount -> {
            refereeAccount.earnPoints(REFEREE_BONUS_POINTS, "referral_bonus");
            loyaltyAccountRepository.save(refereeAccount);
        });

        log.info("Referral processed successfully. Referrer: {}, Referee: {}", 
            referrerAccount.getUserId(), newUserId);

        // Publish event
        publishLoyaltyEvent("loyalty.referral.processed", referrerAccount, REFERRAL_BONUS_POINTS, "referral");

        return loyaltyMapper.toResponse(referrerAccount);
    }

    /**
     * Awards an achievement to a user.
     *
     * @param userId      the user ID
     * @param achievement the achievement to award
     * @return the updated loyalty account response
     */
    @Transactional
    public LoyaltyAccountResponse awardAchievement(String userId, AchievementVO achievement) {
        log.info("Awarding achievement {} to user: {}", achievement.getAchievementId(), userId);

        LoyaltyAccount account = loyaltyAccountRepository.findByUserId(userId)
            .orElseThrow(() -> new LoyaltyAccountNotFoundException(userId));

        achievement.setUnlockedAt(Instant.now());
        account.addAchievement(achievement);

        LoyaltyAccount updated = loyaltyAccountRepository.save(account);
        log.info("Achievement awarded successfully");

        return loyaltyMapper.toResponse(updated);
    }

    /**
     * Awards a badge to a user.
     *
     * @param userId the user ID
     * @param badge  the badge to award
     * @return the updated loyalty account response
     */
    @Transactional
    public LoyaltyAccountResponse awardBadge(String userId, BadgeVO badge) {
        log.info("Awarding badge {} to user: {}", badge.getBadgeId(), userId);

        LoyaltyAccount account = loyaltyAccountRepository.findByUserId(userId)
            .orElseThrow(() -> new LoyaltyAccountNotFoundException(userId));

        badge.setEarnedAt(Instant.now());
        account.addBadge(badge);

        LoyaltyAccount updated = loyaltyAccountRepository.save(account);
        log.info("Badge awarded successfully");

        return loyaltyMapper.toResponse(updated);
    }

    /**
     * Gets accounts by tier with pagination.
     *
     * @param tier     the loyalty tier
     * @param pageable pagination information
     * @return page of loyalty accounts
     */
    public Page<LoyaltyAccountResponse> getAccountsByTier(LoyaltyTier tier, Pageable pageable) {
        log.debug("Fetching accounts for tier: {}", tier);

        return loyaltyAccountRepository.findByTier(tier, pageable)
            .map(loyaltyMapper::toResponse);
    }

    /**
     * Gets the leaderboard (top accounts by points).
     *
     * @param pageable pagination information
     * @return page of top loyalty accounts
     */
    public Page<LoyaltyAccountResponse> getLeaderboard(Pageable pageable) {
        log.debug("Fetching loyalty leaderboard");

        return loyaltyAccountRepository.findTopAccountsByPoints(pageable)
            .map(loyaltyMapper::toResponse);
    }

    // ==================== Helper Methods ====================

    private String generateReferralCode() {
        String code;
        do {
            code = "FD" + UUID.randomUUID().toString().substring(0, 8).toUpperCase();
        } while (loyaltyAccountRepository.existsByReferralCode(code));
        return code;
    }

    private void publishLoyaltyEvent(String topic, LoyaltyAccount account, long points, String source) {
        try {
            Map<String, Object> event = new HashMap<>();
            event.put("eventId", UUID.randomUUID().toString());
            event.put("eventType", topic);
            event.put("timestamp", Instant.now().toString());
            event.put("userId", account.getUserId());
            event.put("accountId", account.getId());
            event.put("points", points);
            event.put("source", source);
            event.put("tier", account.getTier());
            event.put("currentPoints", account.getCurrentPoints());

            kafkaTemplate.send(topic, account.getUserId(), event);
            log.debug("Published event to {}", topic);
        } catch (Exception e) {
            log.error("Failed to publish event to {}", topic, e);
        }
    }
}
